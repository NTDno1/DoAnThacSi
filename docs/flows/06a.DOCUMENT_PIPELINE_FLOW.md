## 1. Tổng quan Pipeline

### 1.1. Giới thiệu

Document Processing Pipeline là hệ thống xử lý tài liệu end-to-end, chịu trách nhiệm tiếp nhận tài liệu từ người dùng, trích xuất nội dung, tạo vector embeddings, và lưu trữ để phục vụ cho semantic search. Pipeline này đảm bảo tài liệu được xử lý một cách tự động, hiệu quả và có khả năng mở rộng.

Pipeline được thiết kế theo nguyên tắc:

- **Asynchronous Processing**: Sử dụng background workers để xử lý không đồng bộ, không blocking UI
- **Fault Tolerance**: Cơ chế retry và dead letter queue đảm bảo không mất job
- **Scalability**: Kiến trúc có thể scale ngang bằng cách thêm workers
- **Observability**: Logging và monitoring ở mọi bước

### Bước 1: Upload (Frontend → Backend)

#### 3.1.1. Multipart Upload

Frontend sử dụng FormData để upload file lên server qua endpoint `POST /api/documents/upload`.

```csharp
// Controllers/DocumentUploadController.cs
[ApiController]
[Route("api/[controller]")]
public class DocumentUploadController : ControllerBase
{
    private readonly IDocumentService _documentService;
    private readonly ILogger<DocumentUploadController> _logger;

    [HttpPost("upload")]
    [RequestSizeLimit(104857600)] // 100MB
    [RequestFormLimits(MultipartBodyLengthLimit = 104857600)]
    public async Task<ActionResult<UploadResponse>> Upload(
        IFormFile file,
        [FromForm] string? categoryId = null,
        [FromForm] string? description = null,
        CancellationToken cancellationToken = default)
    {
        // Validate file
        var validationResult = ValidateFile(file);
        if (!validationResult.IsValid)
        {
            return BadRequest(new { error = validationResult.ErrorMessage });
        }

        // Calculate file hash
        using var sha256 = SHA256.Create();
        using var stream = file.OpenReadStream();
        var hashBytes = await sha256.ComputeHashAsync(stream, cancellationToken);
        var fileHash = Convert.ToHexString(hashBytes);

        // Check for duplicate
        var existingDoc = await _documentService.FindByHashAsync(fileHash, cancellationToken);
        if (existingDoc != null)
        {
            return Conflict(new { 
                message = "Tài liệu đã tồn tại", 
                existingDocumentId = existingDoc.Id 
            });
        }

        // Create document record
        var document = await _documentService.CreateDocumentAsync(new CreateDocumentRequest
        {
            FileName = file.FileName,
            ContentType = file.ContentType,
            FileSize = file.Length,
            FileHash = fileHash,
            CategoryId = categoryId,
            Description = description,
            OriginalFileName = file.FileName
        }, cancellationToken);

        // Save to MinIO
        var minioPath = await _documentService.SaveToMinIOAsync(
            file, 
            document.Id, 
            cancellationToken);

        // Create processing job
        var jobId = await _documentService.QueueProcessingJobAsync(
            document.Id, 
            file.ContentType,
            cancellationToken);

        return Accepted(new UploadResponse
        {
            DocumentId = document.Id,
            JobId = jobId,
            Status = "queued",
            Message = "Tài liệu đã được upload và đang chờ xử lý"
        });
    }

    private FileValidationResult ValidateFile(IFormFile file)
    {
        // Check file size
        if (file.Length > 104857600) // 100MB
        {
            return new FileValidationResult(false, "File vượt quá kích thước cho phép (100MB)");
        }

        // Check file extension
        var allowedExtensions = new[] { ".pdf", ".docx", ".txt", ".xlsx", ".pptx", ".png", ".jpg", ".jpeg", ".tiff" };
        var extension = Path.GetExtension(file.FileName).ToLowerInvariant();
        if (!allowedExtensions.Contains(extension))
        {
            return new FileValidationResult(false, $"Định dạng {extension} không được hỗ trợ");
        }

        // Check MIME type
        var allowedTypes = new[]
        {
            "application/pdf",
            "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
            "text/plain",
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
            "application/vnd.openxmlformats-officedocument.presentationml.presentation",
            "image/png",
            "image/jpeg",
            "image/tiff"
        };
        if (!allowedTypes.Contains(file.ContentType))
        {
            return new FileValidationResult(false, "Loại file không được hỗ trợ");
        }

        return new FileValidationResult(true, null);
    }
}
```

#### 3.1.2. Client-side Validation (TypeScript)

```typescript
// services/document-upload.service.ts
export class DocumentUploadService {
  private readonly MAX_FILE_SIZE = 100 * 1024 * 1024; // 100MB
  private readonly ALLOWED_EXTENSIONS = [
    '.pdf', '.docx', '.txt', '.xlsx', '.pptx', 
    '.png', '.jpg', '.jpeg', '.tiff'
  ];
  private readonly ALLOWED_TYPES = [
    'application/pdf',
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    'text/plain',
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    'application/vnd.openxmlformats-officedocument.presentationml.presentation',
    'image/png',
    'image/jpeg',
    'image/tiff'
  ];

  validateFile(file: File): ValidationResult {
    const errors: string[] = [];

    // Check file size
    if (file.size > this.MAX_FILE_SIZE) {
      errors.push(`File "${file.name}" vượt quá 100MB`);
    }

    // Check extension
    const extension = '.' + file.name.split('.').pop()?.toLowerCase();
    if (!this.ALLOWED_EXTENSIONS.includes(extension)) {
      errors.push(`Định dạng ${extension} không được hỗ trợ`);
    }

    // Check MIME type
    if (!this.ALLOWED_TYPES.includes(file.type)) {
      errors.push(`Loại file không hợp lệ: ${file.type}`);
    }

    return {
      isValid: errors.length === 0,
      errors
    };
  }

  async uploadFile(
    file: File,
    categoryId?: string,
    onProgress?: (progress: number) => void
  ): Promise<UploadResponse> {
    const validation = this.validateFile(file);
    if (!validation.isValid) {
      throw new Error(validation.errors.join(', '));
    }

    const formData = new FormData();
    formData.append('file', file);
    if (categoryId) {
      formData.append('categoryId', categoryId);
    }

    const response = await fetch('/api/documentupload/upload', {
      method: 'POST',
      body: formData,
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.message || 'Upload thất bại');
    }

    return response.json();
  }
}
```

#### 3.2.1. Queue Configuration với BullMQ

```csharp
// Services/Queueing/DocumentQueueService.cs
public class DocumentQueueService : IDocumentQueueService
{
    private readonly IQueue _queue;
    private readonly ILogger<DocumentQueueService> _logger;

    public DocumentQueueService(
        IQueue queue,
        ILogger<DocumentQueueService> logger)
    {
        _queue = queue;
        _logger = logger;
    }

    public async Task<string> EnqueueProcessingJobAsync(
        Guid documentId,
        string contentType,
        CancellationToken cancellationToken = default)
    {
        var jobType = DetermineJobType(contentType);
        
        var job = new DocumentProcessingJob
        {
            DocumentId = documentId,
            JobType = jobType,
            Priority = GetPriority(contentType),
            CreatedAt = DateTime.UtcNow,
            Attempts = 0,
            MaxAttempts = 3
        };

        var jobId = await _queue.AddAsync(jobType, job, new JobOptions
        {
            Priority = job.Priority,
            Attempts = 3,
            Backoff = new BackoffOptions
            {
                Type = BackoffType.Exponential,
                Delay = 5000 // 5 seconds initial delay
            },
            RemoveOnComplete = false,
            RemoveOnFail = false
        });

        _logger.LogInformation(
            "Created processing job {JobId} for document {DocumentId}, type: {JobType}",
            jobId, documentId, jobType);

        return jobId;
    }

    private static string DetermineJobType(string contentType)
    {
        return contentType switch
        {
            "application/pdf" => JobTypes.ProcessPdf,
            "application/vnd.openxmlformats-officedocument.wordprocessingml.document" => JobTypes.ProcessDocx,
            "text/plain" => JobTypes.ProcessText,
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" => JobTypes.ProcessExcel,
            "application/vnd.openxmlformats-officedocument.presentationml.presentation" => JobTypes.ProcessPptx,
            "image/png" or "image/jpeg" or "image/tiff" => JobTypes.ProcessImage,
            _ => JobTypes.ProcessText
        };
    }

    private static int GetPriority(string contentType)
    {
        // Lower number = higher priority
        return contentType switch
        {
            "text/plain" => 1,  // Highest - plain text, fastest processing
            "application/pdf" => 2,
            "application/vnd.openxmlformats-officedocument.wordprocessingml.document" => 2,
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" => 3,
            "application/vnd.openxmlformats-officedocument.presentationml.presentation" => 3,
            "image/png" or "image/jpeg" or "image/tiff" => 4, // OCR - slowest
            _ => 5
        };
    }
}
```

#### 3.2.2. Worker Registration

```csharp
// Workers/DocumentProcessingWorker.cs
public class DocumentProcessingWorker : BackgroundService
{
    private readonly IWorker _worker;
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<DocumentProcessingWorker> _logger;

    public DocumentProcessingWorker(
        IWorker worker,
        IServiceScopeFactory scopeFactory,
        ILogger<DocumentProcessingWorker> logger)
    {
        _worker = worker;
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // Register handlers for each job type
        _worker.AddHandler<DocumentProcessingJob>(ProcessDocumentAsync);

        // Start worker
        await _worker.RunAsync(stoppingToken);
    }

    private async Task ProcessDocumentAsync(DocumentProcessingJob job, JobResult result)
    {
        using var scope = _scopeFactory.CreateScope();
        var processingService = scope.ServiceProvider
            .GetRequiredService<IDocumentProcessingService>();

        try
        {
            _logger.LogInformation(
                "Processing document {DocumentId} with job type {JobType}",
                job.DocumentId, job.JobType);

            await processingService.ProcessDocumentAsync(job.DocumentId, stoppingToken);
            
            result.SetCompleted();
        }
        catch (Exception ex)
        {
```

```csharp
                    PageNumber = pageNumber++,
                    Text = textBuilder.ToString(),
                    HasImages = false,
                    IsScanned = false

                textBuilder.Clear();
            }

            textBuilder.AppendLine(text);
            textBuilder.AppendLine();
        }

        // Add remaining content
        if (textBuilder.Length > 0)
        {
            pages.Add(new PageContent
            {
                PageNumber = pageNumber,
                Text = textBuilder.ToString(),
                HasImages = false,
                IsScanned = false
            });
        }

        {
            Success = true,
            Pages = pages,
            TotalPages = pages.Count,
```

```csharp
public class TextCleaningService : ITextCleaningService

    public CleanedTextResult CleanText(ExtractionResult extractionResult)
    {
        var allCleanedText = new StringBuilder();

        foreach (var page in extractionResult.Pages)
        {
            var cleanedPageText = CleanPageText(page.Text);
            
            allCleanedText.AppendLine($"--- Trang {page.PageNumber} ---");
            allCleanedText.AppendLine(cleanedPageText);
            allCleanedText.AppendLine();
        }

        {

            OriginalLength = extractionResult.Pages.Sum(p => p.Text.Length),
            CleanedLength = allCleanedText.Length,
            Language = DetectLanguage(allCleanedText.ToString())
        };
    }

    private string CleanPageText(string text)
    {
        if (string.IsNullOrWhiteSpace(text))
            return string.Empty;

        var result = text;

        // 1. Fix encoding issues
        result = FixEncoding(result);

        // 2. Normalize Unicode (NFC to NFD or vice versa)
        result = NormalizeUnicode(result);

        // 3. Remove extra whitespace
        result = Regex.Replace(result, @"\s+", " ");

        // 4. Fix line breaks (normalize to \n)
        result = Regex.Replace(result, @"[\r\n]+", "\n");

        // 5. Remove control characters (except \n, \t)
        result = Regex.Replace(result, @"[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]", "");

        // 6. Remove multiple spaces within lines
        result = Regex.Replace(result, @"[^\S\n]+", " ");

        // 7. Preserve paragraph structure
        result = PreserveParagraphStructure(result);

        // 8. Handle special characters
        result = HandleSpecialCharacters(result);

        // 9. Trim whitespace
        result = result.Trim();

        return result;
    }

    private string FixEncoding(string text)
    {
        // Common encoding fixes for Vietnamese documents
        var replacements = new Dictionary<string, string>
        {
            // Fix common OCR/encoding errors
            { "â€" , "\"" },
            { "â€\"", "\"" },
            { "â€œ", "\"" },
            { "â€"", "\"" },
            { "â€˜", "'" },
            { "â€™", "'" },
            { "â€"", "'" },
            { "â€"", "'" },
            { "â€" , "..." },
            { "â€"", "..." },
            { "â€"¢", "•" },
            { "Ã¡", "á" },
            { "Ã ", "à" },
            { "Ã¢", "â" },
            { "Ã£", "ã" },
            { "Äƒ", "ă" },
            { "Ã©", "é" },
            { "Ã¨", "è" },
            { "Ãª", "ê" },
            { "Ã­", "í" },
            { "Ã¬", "ì" },
            { "Ã®", "î" },
            { "Ãµ", "õ" },
            { "Ã³", "ó" },
            { "Ã²", "ò" },
            { "Ã´", "ô" },
            { "Æ¡", "ơ" },
            { "Ãº", "ú" },
            { "Ã¹", "ù" },
            { "Ã»", "û" },
            { "Æ°", "ư" },
            { "Ã½", "ý" },
            { "á»", "ỵ" },
        };

        foreach (var replacement in replacements)
        {
            text = text.Replace(replacement.Key, replacement.Value);
        }

    }

    private string NormalizeUnicode(string text)
    {
        // Normalize to NFC form (Canonical Decomposition, followed by Canonical Composition)
        return text.Normalize(NormalizationForm.FormC);
    }

    private string PreserveParagraphStructure(string text)
    {
        // Ensure blank lines between paragraphs
        var lines = text.Split('\n');
        var result = new List<string>();
        var emptyLineCount = 0;

        foreach (var line in lines)
        {
            if (string.IsNullOrWhiteSpace(line))
            {
                emptyLineCount++;
                if (emptyLineCount <= 1) // Keep single empty line
                {
                    result.Add(line);
                }
            }
            else
            {
                emptyLineCount = 0;
                result.Add(line.Trim());
            }
        }

        return string.Join("\n", result);
    }

    private string HandleSpecialCharacters(string text)
    {
        // Preserve Vietnamese punctuation
        var result = text;

        // Normalize smart quotes to standard
        result = Regex.Replace(result, "[""\"]", "\"");
        result = Regex.Replace(result, "['\']", "'");

        // Preserve em-dash and en-dash
        result = result.Replace("–", "-").Replace("—", "-");

        return result;
    }

    private string DetectLanguage(string text)
    {
        // Simple language detection based on common Vietnamese characters
        var vietnameseChars = new[] { 'đ', 'Đ', 'á', 'à', 'ả', 'ã', 'ạ', 'ă', 'ắ', 'ằ', 'ẳ', 'ẵ', 'ặ' };
        
        var vietnameseCount = text.Count(c => vietnameseChars.Contains(c));
        var totalChars = text.Count(char.IsLetter);
        
        if (totalChars == 0) return "unknown";
        
        var vietnameseRatio = (double)vietnameseCount / totalChars;
        
        return vietnameseRatio > 0.05 ? "vietnamese" : "mixed";
    }
```

```csharp
// Services/Processing/TextChunkingService.cs
public interface ITextChunkingService
{
    List<TextChunk> ChunkText(
        string text, 
        ChunkingOptions options,
        DocumentMetadata metadata);
}

public class TextChunkingService : ITextChunkingService
{
    private readonly ILogger<TextChunkingService> _logger;

    public List<TextChunk> ChunkText(
        string text, 
        ChunkingOptions options,
        DocumentMetadata metadata)
    {
        return options.Strategy switch
        {
            ChunkingStrategy.RecursiveCharacter => ChunkWithRecursiveCharacter(text, options, metadata),
            ChunkingStrategy.Semantic => ChunkWithSemantic(text, options, metadata),
            ChunkingStrategy.FixedSize => ChunkWithFixedSize(text, options, metadata),
            ChunkingStrategy.RecursiveWithOverlap => ChunkWithRecursiveAndOverlap(text, options, metadata),
            _ => ChunkWithRecursiveCharacter(text, options, metadata)
        };
    }

    /// <summary>
    /// Strategy 1: Recursive Character Splitting (Recommended)
    /// Split by: \n\n > \n > . > ! > ? > , > space
    /// Target: 500-1000 tokens per chunk
    /// </summary>
    private List<TextChunk> ChunkWithRecursiveCharacter(
        string text, 
        ChunkingOptions options,
        DocumentMetadata metadata)
    {
        var chunks = new List<TextChunk>();
        var separators = new[] { "\n\n", "\n", ". ", "! ", "? ", ", ", " " };
        
        var paragraphs = text.Split(new[] { "\n\n", "\r\n\r\n" }, StringSplitOptions.RemoveEmptyEntries);
        var currentChunk = new StringBuilder();
        var chunkIndex = 0;

        foreach (var paragraph in paragraphs)
        {
            var trimmedParagraph = paragraph.Trim();
            
            if (string.IsNullOrEmpty(trimmedParagraph))
                continue;

            // If single paragraph is too large, split further
            if (EstimateTokens(trimmedParagraph) > options.MaxTokens * 1.5)
            {
                // Flush current chunk first
                if (currentChunk.Length > 0)
                {
                    chunks.Add(CreateChunk(
                        currentChunk.ToString(), 
                        chunkIndex++, 
                        metadata));
                    currentChunk.Clear();
                }

                // Split by sentences
                var sentences = SplitBySentences(trimmedParagraph);
                foreach (var sentence in sentences)
                {
                    if (EstimateTokens(currentChunk.ToString() + sentence) > options.MaxTokens)
                    {
                        if (currentChunk.Length > 0)
                        {
                            chunks.Add(CreateChunk(
                                currentChunk.ToString(), 
                                chunkIndex++, 
                                metadata));
                            currentChunk.Clear();
                        }
                    }
                    currentChunk.Append(sentence).Append(" ");
                }
            }
            else if (EstimateTokens(currentChunk.ToString() + trimmedParagraph) > options.MaxTokens)
            {
                // Current chunk is full, create new chunk
                chunks.Add(CreateChunk(
                    currentChunk.ToString().Trim(), 
                    chunkIndex++, 
                    metadata));
                currentChunk.Clear();
                currentChunk.Append(trimmedParagraph).Append("\n\n");
            }
            else
            {
                currentChunk.Append(trimmedParagraph).Append("\n\n");
            }
        }

        // Add remaining content
        if (currentChunk.Length > 0)
        {
            chunks.Add(CreateChunk(
                currentChunk.ToString().Trim(), 
                chunkIndex, 
                metadata));
        }

        return chunks;
    }

    /// <summary>
    /// Strategy 2: Recursive with Overlap (Recommended for RAG)
    /// Target overlap: 10-20% (50-100 tokens)
    /// </summary>
    private List<TextChunk> ChunkWithRecursiveAndOverlap(
        string text, 
        ChunkingOptions options,
        DocumentMetadata metadata)
    {
        // First, create base chunks without overlap
        var baseChunks = ChunkWithRecursiveCharacter(text, options, metadata);
        
        if (baseChunks.Count < 2 || options.OverlapTokens <= 0)
            return baseChunks;

        var chunksWithOverlap = new List<TextChunk>();
        var overlapRatio = (double)options.OverlapTokens / options.MaxTokens;
        var overlapChars = (int)(options.MaxTokens * 4 * overlapRatio); // ~4 chars per token

        for (int i = 0; i < baseChunks.Count; i++)
        {
            var baseChunk = baseChunks[i];
            
            // Calculate overlap from previous chunk
            string prefix = "";
            if (i > 0)
            {
                var prevChunk = baseChunks[i - 1].Text;
                prefix = GetLastNCharacters(prevChunk, overlapChars);
            }

            // Calculate suffix for next chunk (if not last)
            string suffix = "";
            if (i < baseChunks.Count - 1)
            {
                var nextChunk = baseChunks[i + 1].Text;
                suffix = GetFirstNCharacters(nextChunk, overlapChars);
            }

            var enrichedChunk = new TextChunk
            {
                Id = Guid.NewGuid(),
                DocumentId = metadata.DocumentId,
                Text = baseChunk.Text,
                Prefix = prefix,
                Suffix = suffix,
                ChunkIndex = i,
                TotalChunks = baseChunks.Count,
                StartPosition = baseChunk.StartPosition,
                EndPosition = baseChunk.EndPosition,
                TokenCount = EstimateTokens(baseChunk.Text),
                Metadata = metadata
            };

            chunksWithOverlap.Add(enrichedChunk);
        }

        return chunksWithOverlap;
    }

    /// <summary>
    /// Strategy 3: Semantic Chunking (Using LLM to identify topic boundaries)
    /// </summary>
    private async Task<List<TextChunk>> ChunkWithSemanticAsync(
        string text, 
        ChunkingOptions options,
        DocumentMetadata metadata)
    {
        // This strategy uses LLM to identify natural topic boundaries
        // More accurate but slower and more expensive
        
        var chunks = new List<TextChunk>();
        
        // First, split into paragraphs
        var paragraphs = text.Split(new[] { "\n\n", "\r\n\r\n" }, StringSplitOptions.RemoveEmptyEntries);
        
        var currentGroup = new List<string>();
        var currentTokens = 0;

        foreach (var paragraph in paragraphs)
        {
            var paragraphTokens = EstimateTokens(paragraph);

            if (currentTokens + paragraphTokens > options.MaxTokens && currentGroup.Count > 0)
            {
                // Create chunk from current group
                var chunkText = string.Join("\n\n", currentGroup);
                chunks.Add(CreateChunk(chunkText, chunks.Count, metadata));
                
                // Start new group
                currentGroup = new List<string>();
                currentTokens = 0;
            }

            currentGroup.Add(paragraph.Trim());
            currentTokens += paragraphTokens;
        }

        // Add remaining content
        if (currentGroup.Count > 0)
        {
            chunks.Add(CreateChunk(
                string.Join("\n\n", currentGroup), 
                chunks.Count, 
                metadata));
        }

        return chunks;
    }

    private List<string> SplitBySentences(string text)
    {
        // Vietnamese-aware sentence splitting
        var sentences = new List<string>();
        
        // Split by common sentence endings
        var pattern = @"(?<=[.!?])\s+";
        var parts = Regex.Split(text, pattern);
        
        var currentSentence = new StringBuilder();
        
        foreach (var part in parts)
        {
            currentSentence.Append(part);
            
            // Check if this looks like end of sentence
            if (part.EndsWith(".") || part.EndsWith("!") || part.EndsWith("?") ||
                part.EndsWith(".") || part.EndsWith("!") || part.EndsWith("?"))
            {
                sentences.Add(currentSentence.ToString());
                currentSentence.Clear();
            }
        }
        
        if (currentSentence.Length > 0)
        {
            sentences.Add(currentSentence.ToString());
        }

        return sentences;
    }

    private int EstimateTokens(string text)
    {
        // Rough estimation: ~4 characters per token for Vietnamese/English mixed
        // More accurate: use tiktoken or similar
        return text.Length / 4;
    }

    private TextChunk CreateChunk(string text, int index, DocumentMetadata metadata)
    {
        return new TextChunk
        {
            Id = Guid.NewGuid(),
            DocumentId = metadata.DocumentId,
            Text = text,
            ChunkIndex = index,
            TotalChunks = 0, // Will be updated later
            StartPosition = 0, // Calculate based on text position
            EndPosition = text.Length,
            TokenCount = EstimateTokens(text),
            Metadata = metadata,
            CreatedAt = DateTime.UtcNow
        };
    }
}
```

```csharp
// Services/Embedding/EmbeddingService.cs
public interface IEmbeddingService

public class EmbeddingService : IEmbeddingService
{
    private readonly HttpClient _httpClient;
    private readonly IRedisCacheService _cacheService;
    private readonly ILogger<EmbeddingService> _logger;
    private readonly IEmbeddingMetricsService _metricsService;
    
    private const string EmbeddingModel = "text-embedding-3-small";
    private const int EmbeddingDimensions = 1536;
    private const int MaxBatchSize = 1000; // OpenAI limit
    private const int RecommendedBatchSize = 100;

    public async Task<EmbeddingResult> GenerateEmbeddingAsync(
        string text, 
        CancellationToken cancellationToken = default)
    {
        // Check cache first
        var cacheKey = GenerateCacheKey(text);
        var cached = await _cacheService.GetAsync<EmbeddingResult>(cacheKey);
        if (cached != null)
        {
            _logger.LogDebug("Cache hit for embedding");
            return cached;
        }

        // Generate embedding
        var response = await CallEmbeddingApiAsync(text, cancellationToken);
        
        var result = new EmbeddingResult
        {
            Embedding = response.Data.First().Embedding,
            Model = response.Model,
            TokenUsage = response.Usage.TotalTokens,
            Dimensions = response.Data.First().Embedding.Length
        };

        // Cache the result
        await _cacheService.SetAsync(cacheKey, result, TimeSpan.FromDays(30));
        
        // Track metrics
        await _metricsService.TrackEmbeddingAsync(result.TokenUsage);

        return result;
    }

    public async Task<List<EmbeddingResult>> GenerateBatchEmbeddingsAsync(
        List<string> texts, 
        int batchSize = RecommendedBatchSize,
        CancellationToken cancellationToken = default)
    {
        var results = new List<EmbeddingResult>();
        var batches = texts.Chunk(batchSize).ToList();

        _logger.LogInformation(
            "Processing {TotalTexts} texts in {BatchCount} batches",
            texts.Count, batches.Count);

        foreach (var batch in batches)
        {
            cancellationToken.ThrowIfCancellationRequested();

            try
            {
                var batchResults = await ProcessBatchAsync(batch.ToList(), cancellationToken);
                results.AddRange(batchResults);

                _logger.LogInformation(
                    "Processed batch: {Processed}/{Total} texts",
                    results.Count, texts.Count);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error processing batch");
                
                // Retry with exponential backoff
                var retryCount = 0;
                var maxRetries = 3;
                
                while (retryCount < maxRetries)
                {
                    try
                    {
                        await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, retryCount)), cancellationToken);
                        var retryResults = await ProcessBatchAsync(batch.ToList(), cancellationToken);
                        results.AddRange(retryResults);
                        break;
                    }
                    catch
                    {
                        retryCount++;
                        if (retryCount >= maxRetries)
                        {
                            throw;
                        }
                    }
                }
            }
        }

        return results;
    }

    private async Task<List<EmbeddingResult>> ProcessBatchAsync(
        List<string> texts, 
        CancellationToken cancellationToken)
    {
        var response = await CallBatchEmbeddingApiAsync(texts, cancellationToken);
        
        var results = response.Data
            .OrderBy(d => d.Index)
            .Select(d => new EmbeddingResult
            {
                Embedding = d.Embedding,
                Model = response.Model,
                TokenUsage = response.Usage.TotalTokens / texts.Count,
                Dimensions = d.Embedding.Length
            })
            .ToList();

        // Cache individual embeddings
        for (int i = 0; i < texts.Count; i++)
        {
            var cacheKey = GenerateCacheKey(texts[i]);
            await _cacheService.SetAsync(cacheKey, results[i], TimeSpan.FromDays(30));
        }

        return results;
    }
```

```csharp
    public async Task<List<SimilarityResult>> SearchSimilarChunksAsync(
        float[] queryEmbedding, 
        int topK,
        SearchOptions options,
        CancellationToken cancellationToken = default)
    {
        var embeddingString = "[" + string.Join(",", queryEmbedding) + "]";

        var query = $@"
            SELECT 
                dc.id AS ChunkId,
                dc.document_id AS DocumentId,
                dc.text,
                dc.chunk_index AS ChunkIndex,
                d.title AS DocumentTitle,
                d.category_id AS CategoryId,
                d.author AS Author,
                1 - (dc.embedding <=> '{embeddingString}'::vector) AS similarity_score,
                ts_rank(to_tsvector('vietnamese', dc.text), plainto_tsquery('vietnamese', @searchQuery)) AS keyword_rank
            FROM document_chunks dc
            INNER JOIN documents d ON dc.document_id = d.id
            WHERE 
                d.is_indexed = true
                AND d.is_deleted = false
                AND d.status = 'active'";

        // Add filters
        if (options.CategoryIds?.Any() == true)
        {
            query += $" AND d.category_id = ANY(@categoryIds)";
        }

        if (options.Author != null)
        {
            query += " AND d.author = @author";
        }

        if (options.DateFrom.HasValue)
        {
            query += " AND d.created_at >= @dateFrom";
        }

        if (options.DateTo.HasValue)
        {
            query += " AND d.created_at <= @dateTo";
        }

        // Combine vector and keyword search with RRF
        query += @"
            ORDER BY 
                (COALESCE(keyword_rank, 0) * 0.3 + (1 - (dc.embedding <=> '{embeddingString}'::vector)) * 0.7)
                DESC
            LIMIT @topK";

        var parameters = new
        {
            searchQuery = options.Query,
            categoryIds = options.CategoryIds,
            author = options.Author,
            dateFrom = options.DateFrom,
            dateTo = options.DateTo,
            topK = topK
        };

        var results = await _context.Database
            .SqlQueryRaw<SimilarityResult>(query, parameters)
            .ToListAsync(cancellationToken);

        return results;
    }
```

```csharp
// Services/Processing/DocumentProcessingService.cs
public class DocumentProcessingService : IDocumentProcessingService
{
    private readonly ITextExtractionService _textExtractionService;
    private readonly ITextCleaningService _textCleaningService;
    private readonly ITextChunkingService _textChunkingService;
    private readonly IEmbeddingService _embeddingService;
    private readonly IDocumentChunkRepository _chunkRepository;
    private readonly IDocumentRepository _documentRepository;
    private readonly IDocumentNotificationService _notificationService;
    private readonly ILogger<DocumentProcessingService> _logger;

    private const int MaxRetries = 3;
    private const int InitialRetryDelayMs = 1000;

    public async Task ProcessDocumentAsync(
        Guid documentId,
        CancellationToken cancellationToken = default)
    {
        var document = await _documentRepository.GetByIdAsync(documentId);
        if (document == null)
        {
            throw new DocumentNotFoundException(documentId);
        }

        var processingContext = new ProcessingContext
        {
            Document = document,
            StartTime = DateTime.UtcNow
        };

        try
        {
            // Step 1: Update status to processing
            await UpdateDocumentStatusAsync(documentId, "processing", 0, cancellationToken);

            // Step 2: Download from MinIO
            await NotifyProgressAsync(documentId, 5, "Đang tải tài liệu...", cancellationToken);
            var fileStream = await DownloadFileAsync(document, cancellationToken);

            // Step 3: Extract text
            await NotifyProgressAsync(documentId, 10, "Đang trích xuất văn bản...", cancellationToken);
            var extractionResult = await ExecuteWithRetryAsync(
                () => _textExtractionService.ExtractTextAsync(
                    fileStream, document.ContentType, document.FileName, cancellationToken),
                "text extraction",
                cancellationToken);

            // Step 4: Clean text
            await NotifyProgressAsync(documentId, 30, "Đang làm sạch văn bản...", cancellationToken);
            var cleanedResult = _textCleaningService.CleanText(extractionResult);

            // Step 5: Chunk text
            await NotifyProgressAsync(documentId, 50, "Đang chia nhỏ văn bản...", cancellationToken);
            var chunks = _textChunkingService.ChunkText(
                cleanedResult.CleanedText,
                ChunkingConfigurations.ForRag8K,
                CreateMetadata(document));

            // Step 6: Generate embeddings
            await NotifyProgressAsync(documentId, 60, "Đang tạo embeddings...", cancellationToken);
            var embeddings = await GenerateEmbeddingsWithProgressAsync(
                chunks, 
                documentId, 
                cancellationToken);

            // Step 7: Save to database
            await NotifyProgressAsync(documentId, 90, "Đang lưu vào cơ sở dữ liệu...", cancellationToken);
            await SaveChunksWithEmbeddingsAsync(chunks, embeddings, cancellationToken);

            // Step 8: Mark as indexed
            await UpdateDocumentStatusAsync(documentId, "indexed", 100, cancellationToken);

            // Step 9: Notify completion
            var result = new ProcessingResult
            {
                Success = true,
                TotalChunks = chunks.Count,
                ProcessingTimeMs = (long)(DateTime.UtcNow - processingContext.StartTime).TotalMilliseconds
            };

            await _notificationService.NotifyProcessingCompleteAsync(documentId, result);
        }
        catch (Exception ex)
        {
            await HandleProcessingErrorAsync(documentId, ex, processingContext, cancellationToken);
        }
    }

    private async Task<T> ExecuteWithRetryAsync<T>(
        Func<Task<T>> operation,
        string stepName,
        CancellationToken cancellationToken)
    {
        var attempts = 0;
        var delay = InitialRetryDelayMs;

        while (true)
        {
            try
            {
                attempts++;
                return await operation();
            }
            catch (Exception ex) when (attempts < MaxRetries)
            {
                _logger.LogWarning(
                    ex,
                    "Step '{StepName}' failed (attempt {Attempt}/{MaxRetries}). Retrying in {Delay}ms",
                    stepName, attempts, MaxRetries, delay);

                await Task.Delay(delay, cancellationToken);
                delay *= 2; // Exponential backoff
            }
        }
    }

    private async Task HandleProcessingErrorAsync(
        Guid documentId,
        Exception exception,
        ProcessingContext context,
        CancellationToken cancellationToken)
    {
        _logger.LogError(
            exception,
            "Document processing failed for document {DocumentId} after {Attempts} attempts",
            documentId, context.Attempts);

        // Determine if should move to DLQ or mark as failed
        var shouldMoveToDLQ = context.Attempts >= MaxRetries && 
            exception is not DocumentNotFoundException;

        if (shouldMoveToDLQ)
        {
            await MoveToDeadLetterQueueAsync(documentId, exception);
            await UpdateDocumentStatusAsync(
                documentId, 
                "failed_dlq", 
                context.ProgressPercent, 
                exception.Message,
                cancellationToken);
        }
        else
        {
            context.Attempts++;
            await UpdateDocumentStatusAsync(
                documentId, 
                "failed", 
                context.ProgressPercent,
                exception.Message,
                cancellationToken);
        }

        // Send failure notification
        await _notificationService.NotifyProcessingFailedAsync(
            documentId, 
            exception.Message);
    }

    private async Task MoveToDeadLetterQueueAsync(Guid documentId, Exception exception)
    {
        var dlqEntry = new DeadLetterQueueEntry
        {
            Id = Guid.NewGuid(),
            DocumentId = documentId,
            ErrorMessage = exception.Message,
            ErrorStackTrace = exception.StackTrace,
            FailedAt = DateTime.UtcNow,
            RetryCount = context.Attempts,
            RequiresManualIntervention = exception is PdfCorruptedException
        };

        await _deadLetterQueueRepository.AddAsync(dlqEntry);

        _logger.LogWarning(
            "Document {DocumentId} moved to DLQ. Manual intervention required: {Required}",
            documentId, dlqEntry.RequiresManualIntervention);
    }
}
```

```csharp
public class BatchProcessingService

    public async Task<BatchProcessingResult> ProcessBatchAsync(
        List<Guid> documentIds,
        BatchProcessingOptions options,
        CancellationToken cancellationToken = default)
    {
        var result = new BatchProcessingResult();
        var semaphore = new SemaphoreSlim(options.MaxConcurrency);

        var tasks = documentIds.Select(async documentId =>
        {
            await semaphore.WaitAsync(cancellationToken);
            try
            {
                await _processingService.ProcessDocumentAsync(documentId, cancellationToken);
                result.Successes.Add(documentId);
            }
            catch (Exception ex)
            {
                result.Failures.Add(new FailedDocument
                {
                    DocumentId = documentId,
                    Error = ex.Message
                });
            }
            finally
            {
                semaphore.Release();
            }
        });

        await Task.WhenAll(tasks);
        return result;
    }
}
```

### 6.2. Health Checks

```csharp
// Health Checks
public class DocumentProcessingHealthCheck : IHealthCheck
{
    private readonly IQueue _queue;
    private readonly IDatabase _redis;
    private readonly ApplicationDbContext _dbContext;

    public async Task<HealthCheckResult> CheckHealthAsync(
        HealthCheckContext context,
        CancellationToken cancellationToken = default)
    {
        var checks = new Dictionary<string, object>();

        // Check Redis queue
        try
        {
            var queueInfo = await _queue.GetJobCountsAsync();
            checks["queue"] = new
            {
                waiting = queueInfo.Waiting,
                active = queueInfo.Active,
                completed = queueInfo.Completed,
                failed = queueInfo.Failed
            };
        }
        catch (Exception ex)
        {
            checks["queue"] = new { error = ex.Message };
        }

        // Check database
        try
        {
            var canConnect = await _dbContext.Database.CanConnectAsync(cancellationToken);
            checks["database"] = canConnect ? "healthy" : "unhealthy";
        }
        catch (Exception ex)
        {
            checks["database"] = new { error = ex.Message };
        }

        // Check Redis cache
        try
        {
            var ping = await _redis.PingAsync();
            checks["redis"] = new { ping_ms = ping.TotalMilliseconds };
        }
        catch (Exception ex)
        {
            checks["redis"] = new { error = ex.Message };
        }

        // Overall status
        var isHealthy = checks.All(c => 
            c.Value is not string s || s != "unhealthy" || 
            ((dynamic)c.Value)?.error == null);

        return new HealthCheckResult
        {
            Status = isHealthy ? "healthy" : "degraded",
            Checks = checks,
            Timestamp = DateTime.UtcNow
        };
    }
}
```

```yaml
          severity: critical
```
