## 1. Tổng quan RAG Architecture

### 1.1. RAG là gì?

**RAG (Retrieval-Augmented Generation)** là kiến trúc kết hợp giữa tìm kiếm thông tin (retrieval) và sinh text (generation) để tạo ra câu trả lời chính xác và có nguồn trích dẫn. Thay vì dựa hoàn toàn vào knowledge đã được train trong LLM, RAG bổ sung thêm ngữ cảnh từ tài liệu nội bộ.

### 1.2. Tại sao cần RAG cho Enterprise Search?

| Phương pháp | Ưu điểm | Nhược điểm | Phù hợp khi |
|-------------|---------|------------|-------------|
| **RAG** | Có nguồn trích dẫn, cập nhật realtime, chi phí thấp | Phụ thuộc retrieval quality | Cần câu trả lời chính xác với nguồn |
| **Fine-tuning** | Hiểu ngữ cảnh tốt, response tự nhiên | Chi phí cao, cập nhật chậm | Cần style/giọng văn cụ thể |
| **Traditional Search** | Nhanh, đơn giản | Không hiểu ngữ cảnh, keyword matching | Tìm kiếm đơn giản |

### Bước 1: Query Understanding

#### 3.1.1. Query Understanding Service

```csharp
// Services/RAG/QueryUnderstandingService.cs
public interface IQueryUnderstandingService
{
    Task<QueryUnderstandingResult> UnderstandAsync(
        string query,
        ConversationContext? conversationContext,
        CancellationToken cancellationToken = default);
}

public class QueryUnderstandingResult
{
    public string OriginalQuery { get; set; }
    public string RewrittenQuery { get; set; }
    public QueryIntent Intent { get; set; }
    public List<ExtractedEntity> Entities { get; set; }
    public List<string> Keywords { get; set; }
    public string Language { get; set; }
    public double Complexity { get; set; } // 0-1
    public string? Topic { get; set; }
}

public class QueryUnderstandingService : IQueryUnderstandingService
{
    private readonly HttpClient _httpClient;
    private readonly ILogger<QueryUnderstandingService> _logger;

    public async Task<QueryUnderstandingResult> UnderstandAsync(
        string query,
        ConversationContext? conversationContext,
        CancellationToken cancellationToken = default)
    {
        // 1. Language Detection
        var language = DetectLanguage(query);

        // 2. Entity Extraction
        var entities = await ExtractEntitiesAsync(query, cancellationToken);

        // 3. Intent Classification
        var intent = ClassifyIntent(query, entities);

        // 4. Query Rewriting (for follow-up questions)
        var rewrittenQuery = query;
        if (conversationContext != null)
        {
            rewrittenQuery = await RewriteQueryAsync(
                query, 
                conversationContext, 
                cancellationToken);
        }

        // 5. Keyword Extraction
        var keywords = ExtractKeywords(query);

        // 6. Complexity Analysis
        var complexity = AnalyzeComplexity(query);

        return new QueryUnderstandingResult
        {
            OriginalQuery = query,
            RewrittenQuery = rewrittenQuery,
            Intent = intent,
            Entities = entities,
            Keywords = keywords,
            Language = language,
            Complexity = complexity,
            Topic = DetectTopic(query)
        };
    }

    private QueryIntent ClassifyIntent(string query, List<ExtractedEntity> entities)
    {
        var queryLower = query.ToLowerInvariant();

        // Classification based on patterns
        if (queryLower.Contains("liệt kê") || queryLower.Contains("danh sách") || 
            queryLower.Contains("những gì"))
        {
            return QueryIntent.List;
        }

        if (queryLower.Contains("so sánh") || queryLower.Contains("khác nhau") ||
            queryLower.Contains("hơn"))
        {
            return QueryIntent.Comparison;
        }

        if (queryLower.Contains("tại sao") || queryLower.Contains("vì sao") ||
            queryLower.Contains("lí do"))
        {
            return QueryIntent.Explanation;
        }

        if (queryLower.Contains("làm thế nào") || queryLower.Contains("cách"))
        {
            return QueryIntent.HowTo;
        }

        if (queryLower.Contains("khi nào") || queryLower.Contains("ngày") ||
            queryLower.Contains("thời gian"))
        {
            return QueryIntent.Temporal;
        }

        if (queryLower.Contains("ở đâu") || queryLower.Contains("địa điểm"))
        {
            return QueryIntent.Location;
        }

        if (queryLower.Contains("bao nhiêu") || queryLower.Contains("số lượng"))
        {
            return QueryIntent.Quantity;
        }

        return QueryIntent.Information; // Default
    }

    private async Task<string> RewriteQueryAsync(
        string query,
        ConversationContext conversationContext,
        CancellationToken cancellationToken)
    {
        // Use LLM to rewrite query with context
        var systemPrompt = @"
Bạn là trợ lý rewrite câu hỏi follow-up thành câu hỏi độc lập có thể tìm kiếm được.
Sử dụng ngữ cảnh từ cuộc hội thoại trước để làm rõ câu hỏi.

Ví dụ:
- Follow-up: 'Nó áp dụng cho ai?' → 'Chính sách này áp dụng cho những ai?'
- Follow-up: 'Còn về vận chuyển?' → 'Chính sách vận chuyển là gì?'
";

        var response = await CallLLMAsync(new LLMRequest
        {
            SystemPrompt = systemPrompt,
            UserPrompt = $"Ngữ cảnh trước đó: {conversationContext.Summary}\n\nCâu hỏi hiện tại: {query}",
            MaxTokens = 200
        }, cancellationToken);

        return response.Text ?? query;
    }

    private double AnalyzeComplexity(string query)
    {
        // Simple heuristic for complexity
        var wordCount = query.Split(' ', StringSplitOptions.RemoveEmptyEntries).Length;
        var hasMultipleQuestions = query.Contains("?") || 
            (query.Contains("và") && query.Contains("?"));

        return (wordCount > 20 || hasMultipleQuestions) ? 0.8 : 
               (wordCount > 10 ? 0.5 : 0.2);
    }
}

public enum QueryIntent
{
    Information,    // Tìm thông tin chung
    List,           // Liệt kê items
    Comparison,     // So sánh
    Explanation,    // Giải thích tại sao
    HowTo,          // Hướng dẫn cách làm
    Temporal,       // Về thời gian
    Location,       // Về địa điểm
    Quantity,       // Về số lượng
    Definition,     // Định nghĩa
    Summary         // Tóm tắt
}
```

#### 3.1.2. Conversation Context Management

```csharp
// Services/RAG/ConversationContextService.cs
public class ConversationContextService : IConversationContextService
{
    private readonly IChatHistoryRepository _chatHistoryRepository;
    private readonly int _maxContextMessages = 10;
    private readonly int _maxContextTokens = 4000;

    public async Task<ConversationContext> BuildContextAsync(
        Guid sessionId,
        CancellationToken cancellationToken = default)
    {
        var recentMessages = await _chatHistoryRepository
            .GetRecentMessagesAsync(sessionId, _maxContextMessages, cancellationToken);

        var contextMessages = new List<ChatMessage>();
        var tokenCount = 0;

        // Build context from most recent to oldest
        foreach (var message in recentMessages.OrderByDescending(m => m.CreatedAt))
        {
            var messageTokens = EstimateTokens(message.Content);
            
            if (tokenCount + messageTokens > _maxContextTokens)
                break;

            contextMessages.Insert(0, message);
            tokenCount += messageTokens;
        }

        // Detect topic
        var topic = DetectOngoingTopic(contextMessages);

        return new ConversationContext
        {
            SessionId = sessionId,
            Messages = contextMessages,
            Summary = GenerateSummary(contextMessages),
            Topic = topic,
            StartedAt = contextMessages.FirstOrDefault()?.CreatedAt
        };
    }

    private string? DetectOngoingTopic(List<ChatMessage> messages)
    {
        if (messages.Count == 0) return null;

        // Get topic from first user message or most recent
        var firstUserMessage = messages.FirstOrDefault(m => m.Role == "user");
        if (firstUserMessage != null)
        {
            // Use LLM to extract topic or simple keyword extraction
            return ExtractTopicKeyword(firstUserMessage.Content);
        }

        return null;
    }

    private string GenerateSummary(List<ChatMessage> messages)
    {
        if (messages.Count == 0) return "Cuộc hội thoại mới.";

        // Simple concatenation of user messages
        var userMessages = messages
            .Where(m => m.Role == "user")
            .Take(3)
            .Select(m => m.Content);

        return string.Join(" | ", userMessages);
```

```csharp
        string query,

    {
        // 1. Normalize query
        var normalizedQuery = NormalizeQuery(query);

        // 2. Check cache for repeated queries
        var cacheKey = GenerateCacheKey(normalizedQuery);
        var cached = await _cacheService.GetAsync<float[]>(cacheKey);
        if (cached != null)
        {
            _logger.LogDebug("Query embedding cache hit");
            return cached;
        }

        var embedding = await _embeddingService.GenerateEmbeddingAsync(
            normalizedQuery, 
            cancellationToken);

        // 4. Cache the result (queries are often repeated)
        await _cacheService.SetAsync(
            cacheKey, 
            embedding.Embedding, 
            TimeSpan.FromMinutes(30));

        return embedding.Embedding;
    }

    private string NormalizeQuery(string query)
    {
        // Remove extra whitespace
        var normalized = Regex.Replace(query, @"\s+", " ");
        
        // Lowercase for embedding (case-insensitive)
        normalized = normalized.ToLowerInvariant();

        // Remove punctuation that might affect embedding
        normalized = normalized.Trim();

        return normalized;
    }

    {
        using var sha256 = SHA256.Create();
        var hashBytes = sha256.ComputeHash(Encoding.UTF8.GetBytes(query));
        return $"query_embedding:{Convert.ToHexString(hashBytes)}";
    }
```

```csharp
// Services/RAG/ContextRetrievalService.cs
public interface IContextRetrievalService
{
    Task<List<RetrievedChunk>> RetrieveAsync(
        float[] queryEmbedding,
        RetrievalOptions options,
        CancellationToken cancellationToken = default);
}

public class RetrievalOptions
{
    public string Query { get; set; }
    public int TopK { get; set; } = 5;
    public double VectorWeight { get; set; } = 0.7;
    public double KeywordWeight { get; set; } = 0.3;
    public double SimilarityThreshold { get; set; } = 0.7;
    public List<Guid>? CategoryIds { get; set; }
    public string? Author { get; set; }
    public DateTime? DateFrom { get; set; }
    public DateTime? DateTo { get; set; }
    public Guid? UserId { get; set; }
}

public class ContextRetrievalService : IContextRetrievalService
{
    private readonly IDocumentChunkRepository _chunkRepository;
    private readonly IFullTextSearchService _ftsService;
    private readonly ILogger<ContextRetrievalService> _logger;

    public async Task<List<RetrievedChunk>> RetrieveAsync(
        float[] queryEmbedding,
        RetrievalOptions options,
        CancellationToken cancellationToken = default)
    {
        _logger.LogInformation("Starting retrieval for query with TopK={TopK}", options.TopK);

        // 1. Vector Search
        var vectorResults = await VectorSearchAsync(
            queryEmbedding, 
            options.TopK * 2, // Get more for hybrid scoring
            options,
            cancellationToken);

        // 2. Keyword Search (if query has significant keywords)
        List<RetrievedChunk> keywordResults = new();
        if (HasSignificantKeywords(options.Query))
        {
            keywordResults = await KeywordSearchAsync(
                options.Query, 
                options.TopK * 2,
                options,
                cancellationToken);
        }

        // 3. Hybrid Scoring with RRF
        var hybridResults = PerformRRFHybridScoring(
            vectorResults, 
            keywordResults, 
            options.VectorWeight, 
            options.KeywordWeight);

        // 4. Filter by threshold and take Top-K
        var finalResults = hybridResults
            .Where(r => r.HybridScore >= options.SimilarityThreshold)
            .Take(options.TopK)
            .ToList();

        _logger.LogInformation(
            "Retrieved {Count} chunks (vector: {Vector}, keyword: {Keyword}, hybrid: {Final})",
            finalResults.Count, vectorResults.Count, keywordResults.Count, finalResults.Count);

        return finalResults;
    }

    private async Task<List<RetrievedChunk>> VectorSearchAsync(
        float[] embedding,
        int limit,
        RetrievalOptions options,
        CancellationToken cancellationToken)
    {
        var searchOptions = new SearchOptions
        {
            Query = options.Query,
            CategoryIds = options.CategoryIds,
            Author = options.Author,
            DateFrom = options.DateFrom,
            DateTo = options.DateTo
        };

        var results = await _chunkRepository.SearchSimilarChunksAsync(
            embedding, 
            limit, 
            searchOptions, 
            cancellationToken);

        // Normalize scores to 0-1 range
        var maxScore = results.Any() ? results.Max(r => r.SimilarityScore) : 1;
        
        return results.Select((r, index) => new RetrievedChunk
        {
            ChunkId = r.ChunkId,
            DocumentId = r.DocumentId,
            Text = r.Text,
            ChunkIndex = r.ChunkIndex,
            DocumentTitle = r.DocumentTitle,
            Author = r.Author,
            PageNumber = r.PageNumber,
            SimilarityScore = r.SimilarityScore / maxScore,
            VectorScore = r.SimilarityScore,
            KeywordScore = 0,
            HybridScore = 0
        }).ToList();
    }

    private async Task<List<RetrievedChunk>> KeywordSearchAsync(
        string query,
        int limit,
        RetrievalOptions options,
        CancellationToken cancellationToken)
    {
        var results = await _ftsService.SearchAsync(new FullTextSearchRequest
        {
            Query = query,
            Limit = limit,
            CategoryIds = options.CategoryIds,
            DateFrom = options.DateFrom,
            DateTo = options.DateTo
        }, cancellationToken);

        // Normalize BM25 scores to 0-1 range
        var maxScore = results.Any() ? results.Max(r => r.Score) : 1;

        return results.Select((r, index) => new RetrievedChunk
        {
            ChunkId = r.ChunkId,
            DocumentId = r.DocumentId,
            Text = r.Text,
            ChunkIndex = r.ChunkIndex,
            DocumentTitle = r.DocumentTitle,
            SimilarityScore = 0,
            VectorScore = 0,
            KeywordScore = r.Score / maxScore,
            HybridScore = 0
        }).ToList();
    }

    /// <summary>
    /// Reciprocal Rank Fusion (RRF) - Kết hợp rankings từ multiple retrieval methods
    /// </summary>
    private List<RetrievedChunk> PerformRRFHybridScoring(
        List<RetrievedChunk> vectorResults,
        List<RetrievedChunk> keywordResults,
        double vectorWeight,
        double keywordWeight)
    {
        // Create a dictionary of all chunks
        var allChunks = new Dictionary<Guid, RetrievedChunk>();

        // Add vector results
        foreach (var chunk in vectorResults)
        {
            allChunks[chunk.ChunkId] = chunk;
        }

        // Merge keyword results
        foreach (var chunk in keywordResults)
        {
            if (!allChunks.ContainsKey(chunk.ChunkId))
            {
                allChunks[chunk.ChunkId] = chunk;
            }
        }

        // Calculate RRF scores
        // RRF formula: 1 / (k + rank)^a
        // k = 60 (constant), a = 0 (for standard RRF)
        const int k = 60;

        foreach (var chunk in allChunks.Values)
        {
            var vectorRank = vectorResults.FindIndex(r => r.ChunkId == chunk.ChunkId);
            var keywordRank = keywordResults.FindIndex(r => r.ChunkId == chunk.ChunkId);

            double rrfVector = vectorRank >= 0 ? 1.0 / (k + vectorRank + 1) : 0;
            double rrfKeyword = keywordRank >= 0 ? 1.0 / (k + keywordRank + 1) : 0;

            // Weighted combination
            chunk.HybridScore = (vectorWeight * rrfVector + keywordWeight * rrfKeyword);
            
            // Also consider raw similarity scores
            chunk.HybridScore = chunk.HybridScore * 0.7 + 
                (chunk.VectorScore * vectorWeight + chunk.KeywordScore * keywordWeight) * 0.3;
        }

        return allChunks.Values
            .OrderByDescending(c => c.HybridScore)
            .ToList();
    }

    private bool HasSignificantKeywords(string query)
    {
        // Check if query has enough meaningful words for keyword search
        var words = query.Split(' ', StringSplitOptions.RemoveEmptyEntries)
            .Where(w => w.Length > 2)
            .Where(w => !IsStopWord(w))
            .ToList();

        return words.Count >= 2;
    }

    private bool IsStopWord(string word)
    {
        var stopWords = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "và", "của", "là", "có", "được", "trong", "với", "cho", "không",
            "theo", "về", "để", "từ", "này", "đó", "những", "một", "các",
            "and", "the", "is", "in", "with", "for", "to", "of", "a", "an"
        };

        return stopWords.Contains(word);
    }
}
```

```csharp
// Services/RAG/ReRankingService.cs
public interface IReRankingService
{
    Task<List<RetrievedChunk>> ReRankAsync(
        string query,
        List<RetrievedChunk> chunks,
        ReRankOptions options,
        CancellationToken cancellationToken = default);
}

public class ReRankOptions
{
    public int TopK { get; set; } = 5;
    public bool UseMMR { get; set; } = true;
    public double MMRLambda { get; set; } = 0.5; // Balance relevance vs diversity
    public double MinRelevanceScore { get; set; } = 0.3;
}

public class ReRankingService : IReRankingService
{
    private readonly HttpClient _httpClient;
    private readonly ILogger<ReRankingService> _logger;

    public async Task<List<RetrievedChunk>> ReRankAsync(
        string query,
        List<RetrievedChunk> chunks,
        ReRankOptions options,
        CancellationToken cancellationToken = default)
    {
        if (!chunks.Any())
            return chunks;

        _logger.LogInformation("Re-ranking {Count} chunks", chunks.Count);

        // 1. Cross-encoder scoring
        var crossEncoderScores = await GetCrossEncoderScoresAsync(
            query, 
            chunks.Select(c => c.Text).ToList(), 
            cancellationToken);

        // 2. Update chunks with cross-encoder scores
        for (int i = 0; i < chunks.Count; i++)
        {
            chunks[i].CrossEncoderScore = crossEncoderScores[i];
        }

        // 3. Apply MMR (Maximum Marginal Relevance) for diversity
        if (options.UseMMR)
        {
            chunks = ApplyMMR(chunks, options);
        }
        else
        {
            // Simple sorting by cross-encoder score
            chunks = chunks
                .OrderByDescending(c => c.CrossEncoderScore)
                .ToList();
        }

        // 4. Filter by minimum relevance
        var filtered = chunks
            .Where(c => c.CrossEncoderScore >= options.MinRelevanceScore)
            .Take(options.TopK)
            .ToList();

        // 5. Final score calculation (combine with original hybrid score)
        foreach (var chunk in filtered)
        {
            chunk.FinalScore = chunk.CrossEncoderScore * 0.7 + 
                chunk.HybridScore * 0.3;
        }

        return filtered;
    }

    private async Task<List<double>> GetCrossEncoderScoresAsync(
        string query,
        List<string> texts,
        CancellationToken cancellationToken)
    {
        // Call cross-encoder API (e.g., sentence-transformers cross-encoder)
        var request = new
        {
            query,
            texts
        };

        var response = await _httpClient.PostAsJsonAsync(
            "/api/cross-encoder/scores",
            request,
            cancellationToken);

        if (!response.IsSuccessStatusCode)
        {
            _logger.LogWarning("Cross-encoder failed, using original scores");
            return texts.Select((_, i) => 1.0 / (i + 1)).ToList();
        }

        var result = await response.Content
            .ReadFromJsonAsync<CrossEncoderResponse>(cancellationToken);

        return result?.Scores ?? texts.Select((_, i) => 1.0 / (i + 1)).ToList();
    }

    /// <summary>
    /// Maximum Marginal Relevance (MMR) - Cân bằng relevance và diversity
    /// </summary>
    private List<RetrievedChunk> ApplyMMR(
        List<RetrievedChunk> chunks, 
        ReRankOptions options)
    {
        var selected = new List<RetrievedChunk>();
        var remaining = new List<RetrievedChunk>(chunks);

        while (selected.Count < options.TopK && remaining.Any())
        {
            double maxMMR = double.MinValue;
            RetrievedChunk? bestChunk = null;

            foreach (var chunk in remaining)
            {
                // Relevance: cross-encoder score
                var relevance = chunk.CrossEncoderScore;

                // Diversity: minimum similarity to selected items
                var maxSimilarity = selected.Any() 
                    ? selected.Max(s => CosineSimilarity(chunk.Embedding, s.Embedding))
                    : 0;

                // MMR formula: λ * relevance - (1 - λ) * diversity
                var mmrScore = options.MMRLambda * relevance - 
                    (1 - options.MMRLambda) * maxSimilarity;

                if (mmrScore > maxMMR)
                {
                    maxMMR = mmrScore;
                    bestChunk = chunk;
                }
            }

            if (bestChunk != null)
            {
                selected.Add(bestChunk);
                remaining.Remove(bestChunk);
            }
        }

        return selected;
    }

    private double CosineSimilarity(float[] a, float[] b)
    {
        if (a.Length != b.Length) return 0;

        var dotProduct = a.Zip(b, (x, y) => x * y).Sum();
        var magnitudeA = Math.Sqrt(a.Sum(x => x * x));
        var magnitudeB = Math.Sqrt(b.Sum(x => x * x));

        return magnitudeA * magnitudeB == 0 ? 0 : 
            dotProduct / (magnitudeA * magnitudeB);
    }
}
```

```csharp
    private readonly ILogger<ContextAssemblyService> _logger;

    public async Task<AssembledContext> AssembleAsync(
        List<RetrievedChunk> chunks,
        AssemblingOptions options,
        CancellationToken cancellationToken = default)
    {
        _logger.LogInformation("Assembling context from {Count} chunks", chunks.Count);

        var formattedParts = new List<string>();
        var citations = new List<CitationSource>();
        var totalTokens = 0;

        foreach (var chunk in chunks.OrderByDescending(c => c.FinalScore))
        {
            cancellationToken.ThrowIfCancellationRequested();

            var formattedChunk = FormatChunk(chunk, options);
            var chunkTokens = EstimateTokens(formattedChunk);

            // Check if adding this chunk exceeds limit
            if (totalTokens + chunkTokens > options.MaxTokens)
            {
                _logger.LogDebug(
                    "Context limit reached at {CurrentTokens}/{MaxTokens}", 
                    totalTokens, options.MaxTokens);
                break;
            }

            formattedParts.Add(formattedChunk);
            totalTokens += chunkTokens;

            citations.Add(new CitationSource
            {
                ChunkId = chunk.ChunkId,
                DocumentId = chunk.DocumentId,
                DocumentTitle = chunk.DocumentTitle,
                PageNumber = chunk.PageNumber,
                ChunkIndex = chunk.ChunkIndex,
                RelevanceScore = chunk.FinalScore,
                Excerpt = chunk.Text.Length > 200 
                    ? chunk.Text.Substring(0, 200) + "..." 
                    : chunk.Text
            });
        }

        var assembledContext = new AssembledContext
        {
            FormattedContext = string.Join("\n\n---\n\n", formattedParts),
            Citations = citations,
            TotalTokens = totalTokens,
            TotalChunks = formattedParts.Count
        };

        return assembledContext;
    }

    private string FormatChunk(RetrievedChunk chunk, AssemblingOptions options)
    {
        var sb = new StringBuilder();

        // Header with metadata
        sb.AppendLine("**Tài liệu**: " + chunk.DocumentTitle);
        
        if (options.IncludePageNumbers && chunk.PageNumber.HasValue)
        {
            sb.AppendLine("**Trang**: " + chunk.PageNumber.Value);
        }
        
        if (!string.IsNullOrEmpty(chunk.Author))
        {
            sb.AppendLine("**Tác giả**: " + chunk.Author);
        }
        
        sb.AppendLine("**Độ phù hợp**: " + $"{chunk.FinalScore:P0}");
        sb.AppendLine();

        // Content
        sb.AppendLine(chunk.Text);

        return sb.ToString();
    }

    private int EstimateTokens(string text)
    {
        // Rough estimation for Vietnamese/English mixed
        return text.Length / 4;
```

```csharp
                    new { role = "user", content = request.UserPrompt }
                },
                temperature = request.Temperature,
                max_tokens = request.MaxTokens,
                top_p = request.TopP,
                stream = false
            };

            var response = await _httpClient.PostAsJsonAsync(
                "https://api.openai.com/v1/chat/completions",
                openAIRequest,
                cancellationToken);

            if (!response.IsSuccessStatusCode)
            {
                var error = await response.Content.ReadAsStringAsync(cancellationToken);
                _logger.LogError("LLM API error: {Error}", error);
                throw new LLMException($"LLM API failed: {error}");
            }

            var openAIResponse = await response.Content
                .ReadFromJsonAsync<OpenAIChatResponse>(cancellationToken);

            stopwatch.Stop();

            var result = new LLMResponse
            {
                Content = openAIResponse?.Choices?.FirstOrDefault()?.Message?.Content ?? "",
                Model = openAIResponse?.Model ?? request.Model,
                TotalTokens = openAIResponse?.Usage?.TotalTokens ?? 0,
                PromptTokens = openAIResponse?.Usage?.PromptTokens ?? 0,
                CompletionTokens = openAIResponse?.Usage?.CompletionTokens ?? 0,
                LatencyMs = stopwatch.ElapsedMilliseconds,
                FinishReason = openAIResponse?.Choices?.FirstOrDefault()?.FinishReason
            };

            // Track token usage
            await _tokenService.TrackUsageAsync(new TokenUsage
            {
                Model = request.Model,
                PromptTokens = result.PromptTokens,
                CompletionTokens = result.CompletionTokens,
                Timestamp = DateTime.UtcNow
            });

            return result;
        }
        catch (Exception ex) when (ex is not LLMException)
        {
            stopwatch.Stop();
            _logger.LogError(ex, "LLM generation failed");
            
            // Fallback: try with lower model
            if (request.Model == "gpt-4o")
            {
                request.Model = "gpt-4o-mini";
                return await GenerateAsync(request, cancellationToken);
            }

            throw new LLMException("LLM generation failed after fallback", ex);
        }
    }

    public async IAsyncEnumerable<string> StreamGenerateAsync(
        LLMRequest request,
        CancellationToken cancellationToken = default)
    {
        var openAIRequest = new
        {
            model = request.Model,
            messages = new[]
            {
                new { role = "system", content = request.SystemPrompt },
                new { role = "user", content = request.UserPrompt }
            },
            temperature = request.Temperature,
            max_tokens = request.MaxTokens,
            stream = true
        };

        using var response = await _httpClient.PostAsJsonAsync(
            "https://api.openai.com/v1/chat/completions",
            openAIRequest,
            cancellationToken);

        response.EnsureSuccessStatusCode();

        using var reader = new StreamReader(
            await response.Content.ReadAsStreamAsync(cancellationToken));

        while (!reader.EndOfStream)
        {
            var line = await reader.ReadLineAsync(cancellationToken);
            
            if (line.StartsWith("data: "))
            {
                var data = line.Substring(6);
                if (data == "[DONE]")
                    yield break;

                var chunk = JsonSerializer.Deserialize<StreamChunk>(data);
                var content = chunk?.Choices?.FirstOrDefault()?.Delta?.Content;
                
                if (!string.IsNullOrEmpty(content))
                {
                    yield return content;
                }
            }
        }
    }
```

```csharp
    private List<Citation> ExtractCitations(
        string answer, 
        List<CitationSource> sources)
    {
        var citations = new List<Citation>();
        
        // Find citation markers in answer
        var citationPattern = new Regex(@"\[(\d+)\]");
        var matches = citationPattern.Matches(answer);

        var seenIndices = new HashSet<int>();

        foreach (Match match in matches)
        {
            var index = int.Parse(match.Groups[1].Value) - 1; // Convert to 0-based
            
            if (index >= 0 && index < sources.Count && !seenIndices.Contains(index))
            {
                seenIndices.Add(index);
                var source = sources[index];

                citations.Add(new Citation
                {
                    Number = index + 1,
                    DocumentId = source.DocumentId,
                    DocumentTitle = source.DocumentTitle,
                    ChunkId = source.ChunkId,
                    PageNumber = source.PageNumber,
                    RelevanceScore = source.RelevanceScore,
                    Excerpt = source.Excerpt
                });
            }
        }

        return citations;
    }

    private double CalculateConfidence(
        LLMResponse llmResponse,
        AssembledContext context,
        List<Citation> citations)
    {
        var factors = new List<double>();

        // 1. Citation coverage (should cite sources)
        if (citations.Any())
        {
            factors.Add(0.9); // Good - has citations
        }
        else
        {
            factors.Add(0.5); // Warning - no citations
        }

        // 2. Answer length (too short might be insufficient)
        var wordCount = llmResponse.Content.Split(' ').Length;
        if (wordCount > 20)
        {
            factors.Add(0.9);
        }
        else if (wordCount > 10)
        {
            factors.Add(0.7);
        }
        else
        {
            factors.Add(0.5);
        }

        // 3. Context sufficiency
        if (context.TotalChunks >= 3)
        {
            factors.Add(0.9);
        }
        else if (context.TotalChunks >= 1)
        {
            factors.Add(0.7);
        }
        else
        {
            factors.Add(0.4);
        }

        // 4. Token usage efficiency
        var avgTokensPerChunk = context.TotalTokens / Math.Max(1, context.TotalChunks);
        if (avgTokensPerChunk > 500)
        {
            factors.Add(0.9);
        }
        else
        {
            factors.Add(0.7);
        }

        // Average of all factors
        return factors.Average();
    }

    private string? GenerateFollowUps(
        string answer, 
        QueryUnderstandingResult queryUnderstanding)
    {

                followUps.Add("Có mục nào bạn muốn so sánh không?");
                break;

            case QueryIntent.HowTo:
                followUps.Add("Bạn cần hỗ trợ thêm bước nào?");
                followUps.Add("Có điều kiện tiên quyết nào bạn cần lưu ý?");
                break;

            case QueryIntent.Explanation:
                followUps.Add("Bạn có muốn biết thêm về ngoại lệ không?");
                followUps.Add("Có ví dụ cụ thể nào bạn muốn xem không?");
                break;

            default:
                followUps.Add("Bạn có câu hỏi nào khác không?");
                followUps.Add("Bạn cần tìm thêm thông tin về chủ đề này không?");
                break;
        }

        return string.Join("\n", followUps.Take(2));
    }
```

```csharp
        Guid sessionId,

    {
        var context = await _contextService.BuildContextAsync(sessionId, cancellationToken);
        
        // Detect topic drift
        var topicDrift = DetectTopicDrift(context);

        return new ConversationState
        {
            SessionId = sessionId,
            Messages = context.Messages,
            CurrentTopic = context.Topic,
            TopicDriftCount = topicDrift,
            IsTopicDrifted = topicDrift >= _topicDriftThreshold,
            MessageCount = context.Messages.Count
        };
    }

    private int DetectTopicDrift(ConversationContext context)
    {
        if (context.Messages.Count < 4)
            return 0;

        // Compare topics of recent messages
        var recentMessages = context.Messages
            .Where(m => m.Role == "user")
            .TakeLast(3)
            .Select(m => m.Content)
            .ToList();

        if (recentMessages.Count < 2)
            return 0;

        // Simple keyword-based topic comparison
        var firstTopic = ExtractTopicKeywords(recentMessages.First());
        var lastTopic = ExtractTopicKeywords(recentMessages.Last());

        var overlap = firstTopic.Intersect(lastTopic).Count();
        var total = firstTopic.Union(lastTopic).Count();

        if (total == 0) return 0;

        var similarity = (double)overlap / total;
        
        // If less than 30% overlap, consider topic drifted
        return similarity < 0.3 ? 1 : 0;
    }
}
```

```csharp
            }
```

```csharp
public class CorrectiveRAGService : IRAGService
{
    public async Task<RAGResult> ProcessAsync(
        RAGRequest request,
        CancellationToken cancellationToken = default)
    {
        // 1. Initial retrieval
        var initialChunks = await _retrievalService.RetrieveAsync(
            request.QueryEmbedding,
            request.Options,
            cancellationToken);

        // 2. Check retrieval quality
        var qualityCheck = await CheckRetrievalQualityAsync(
            request.Query, 
            initialChunks, 
            cancellationToken);

        if (!qualityCheck.IsGood)
        {
            // 3. Self-correction: Try alternative queries
            _logger.LogWarning("Initial retrieval quality low, trying corrections...");

            var correctedChunks = await TryCorrectedRetrievalAsync(
                request.Query,
                qualityCheck.Issues,
                cancellationToken);

            if (correctedChunks.Any())
            {
                return await GenerateFinalResponseAsync(
                    request, 
                    correctedChunks, 
                    cancellationToken);
            }
        }

        // 4. Fallback: Use initial results
        return await GenerateFinalResponseAsync(
            request, 
            initialChunks, 
            cancellationToken);
    }

    private async Task<RetrievalQualityCheck> CheckRetrievalQualityAsync(
        string query,
        List<RetrievedChunk> chunks,
        CancellationToken cancellationToken)
    {
        var issues = new List<string>();

        // Check 1: Too few results
        if (chunks.Count < 2)
        {
            issues.Add("Too few results retrieved");
        }

        // Check 2: Low relevance scores
        if (chunks.Any() && chunks.Average(c => c.SimilarityScore) < 0.5)
        {
            issues.Add("Low average relevance score");
        }

        // Check 3: Content doesn't match query intent
        var intentMismatch = await CheckIntentMatchAsync(query, chunks, cancellationToken);
        if (intentMismatch)
        {
            issues.Add("Retrieved content doesn't match query intent");
        }

        return new RetrievalQualityCheck
        {
            IsGood = issues.Count == 0,
            Issues = issues
        };
    }
```

```csharp
public class HallucinationPreventionService : IHallucinationPreventionService

        string answer,
        string context,
        CancellationToken cancellationToken = default)

        check.CitationsFound = ExtractCitations(answer);

        // 2. Claim verification (basic)

        check.ConfidenceThresholdMet = check.UnsupportedClaims.Count == 0;
```

## Tóm tắt

RAG Chatbot Flow là hệ thống hoàn chỉnh bao gồm:

1. **Query Understanding**: Phân tích ý định, trích xuất thực thể, quản lý ngữ cảnh
2. **Retrieval**: Hybrid search kết hợp vector + keyword với RRF
3. **Re-ranking**: Cross-encoder và MMR để cải thiện chất lượng
4. **Generation**: Prompt engineering tối ưu cho RAG
5. **Hallucination Prevention**: Nhiều lớp bảo vệ để đảm bảo câu trả lời chính xác
6. **Conversation Management**: Quản lý cuộc hội thoại, follow-up questions

Hệ thống này được thiết kế để:
- Cung cấp câu trả lời chính xác với nguồn trích dẫn
- Giảm thiểu hallucination ở mọi lớp
- Tối ưu chi phí và hiệu suất
- Có khả năng mở rộng và giám sát
