#### 1.2.2 Index Strategy

```sql
-- Clustered Index (1 per table, usually on PK)
CREATE TABLE documents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- other columns
);

-- Non-clustered Indexes
CREATE INDEX idx_documents_owner_id ON documents(owner_id);
CREATE INDEX idx_documents_department ON documents(department_id);
CREATE INDEX idx_documents_created_at ON documents(created_at DESC);

-- Composite Index (for common query patterns)
CREATE INDEX idx_documents_owner_status 
ON documents(owner_id, is_active) 
WHERE is_active = true;

-- Full-text Search Index
CREATE INDEX idx_chunks_content_fts 
ON document_chunks USING GIN(to_tsvector('vietnamese', content));

-- Vector Index (pgvector)
CREATE INDEX idx_chunks_embedding_hnsw 
ON document_chunks USING hnsw (content_embedding vector_cosine_ops)
WITH (m = 16, ef_construction = 64);
```

#### 1.3.3 Validation

```csharp
// FluentValidation Example
public class CreateDocumentCommandValidator : AbstractValidator<CreateDocumentCommand>
{
    public CreateDocumentCommandValidator()
    {
        RuleFor(x => x.Title)
            .NotEmpty().WithMessage("Title is required")
            .MaximumLength(500).WithMessage("Title cannot exceed 500 characters")
            .Matches(@"^[a-zA-Z0-9\s\p{L}]+$")
            .WithMessage("Title contains invalid characters");
        
        RuleFor(x => x.File)
            .NotNull().WithMessage("File is required")
            .Must(BeValidFileType).WithMessage("Invalid file type")
            .Must(BeWithinSizeLimit).WithMessage("File size exceeds limit");
        
        RuleFor(x => x.Category)
            .NotEmpty().WithMessage("Category is required")
            .Must(BeValidCategory).WithMessage("Invalid category");
    }
    
    private bool BeValidFileType(IFormFile file)
    {
        var allowedTypes = new[] { ".pdf", ".txt" };
        var extension = Path.GetExtension(file.FileName).ToLower();
        return allowedTypes.Contains(extension);
    }
    
    private bool BeWithinSizeLimit(IFormFile file)
    {
        return file.Length <= 50 * 1024 * 1024; // 50MB
    }
}
```

```csharp
// JWT Configuration
public class JwtSettings
{
    public string Secret { get; set; }
    public string Issuer { get; set; }
    public string Audience { get; set; }
    public int AccessTokenExpiryMinutes { get; set; }
    public int RefreshTokenExpiryDays { get; set; }
}

// Token Generation
public class JwtTokenService : ITokenService
{
    public async Task<TokenResult> GenerateTokensAsync(User user)
    {
        var accessToken = GenerateAccessToken(user);
        var refreshToken = GenerateRefreshToken();
        
        await _userRepository.UpdateRefreshTokenAsync(
            user.Id, 
            refreshToken, 
            DateTime.UtcNow.AddDays(7));
        
        return new TokenResult
        {
            AccessToken = accessToken,
            RefreshToken = refreshToken,
            ExpiresIn = _jwtSettings.AccessTokenExpiryMinutes * 60
        };
    }
    
    private string GenerateAccessToken(User user)
    {
        var claims = new[]
        {
            new Claim(JwtRegisteredClaimNames.Sub, user.Id.ToString()),
            new Claim(JwtRegisteredClaimNames.Email, user.Email),
            new Claim(ClaimTypes.Role, user.Role.ToString()),
            new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString())
        };
        
        var key = new SymmetricSecurityKey(
            Encoding.UTF8.GetBytes(_jwtSettings.Secret));
        var creds = new SigningCredentials(
            key, SecurityAlgorithms.HmacSha256);
        
        var token = new JwtSecurityToken(
            issuer: _jwtSettings.Issuer,
            audience: _jwtSettings.Audience,
            claims: claims,
            expires: DateTime.UtcNow.AddMinutes(
                _jwtSettings.AccessTokenExpiryMinutes),
            signingCredentials: creds);
        
        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}
```

#### 1.4.2 Authorization

```csharp
// Policy-based Authorization
public class DocumentAuthorizationHandler : 
    AuthorizationHandler<DocumentOperationRequirement, Document>
{
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        DocumentOperationRequirement requirement,
        Document resource)
    {
        var userId = context.User.GetUserId();
        var userRole = context.User.GetRole();
        
        // Admin can do anything
        if (userRole == UserRole.Admin)
        {
            context.Succeed(requirement);
            return Task.CompletedTask;
        }
        
        // Owner can do anything with their documents
        if (resource.OwnerId == userId)
        {
            context.Succeed(requirement);
            return Task.CompletedTask;
        }
        
        // Public documents can be read by anyone
        if (resource.IsPublic && 
            requirement.Operation == DocumentOperation.Read)
        {
            context.Succeed(requirement);
            return Task.CompletedTask;
        }
        
        // Department documents can be read by department members
        if (requirement.Operation == DocumentOperation.Read &&
            _departmentService.IsMember(userId, resource.DepartmentId))
        {
            context.Succeed(requirement);
            return Task.CompletedTask;
        }
        
        return Task.CompletedTask;
    }
}
```

### 1.5 Performance

#### 1.5.1 pgvector Optimization

```sql
-- HNSW Index Configuration
-- HNSW: Hierarchical Navigable Small World
-- Best for: High accuracy requirements

CREATE INDEX idx_chunks_embedding_hnsw 
ON document_chunks 
USING hnsw (content_embedding vector_cosine_ops)
WITH (
    m = 16,                    -- Connections per layer (higher = more accurate, more memory)
    ef_construction = 64      -- Build time accuracy (higher = slower build, more accurate)
);

-- IVFFlat Index (alternative for very large datasets)
CREATE INDEX idx_chunks_embedding_ivfflat 
ON document_chunks 
USING ivfflat (content_embedding vector_cosine_ops)
WITH (lists = 100);  -- Number of clusters

-- Query with EXPLAIN ANALYZE
EXPLAIN ANALYZE
SELECT 
    id,
    content,
    (content_embedding <=> $1::vector) AS similarity
FROM document_chunks
WHERE (content_embedding <=> $1::vector) < 0.5
ORDER BY content_embedding <=> $1::vector
LIMIT 10;

-- pgvector Query Options
/*
<-> (L2 distance) - Euclidean distance
<=> (Cosine distance) - Cosine similarity
<#> (Negative inner product) - For certain use cases

For semantic search: USE Cosine distance (<=>)
*/
```

```sql
SELECT id, (embedding <-> query_embedding) AS l2_distance
FROM chunks ORDER BY embedding <-> query_embedding LIMIT 5;

-- Cosine Distance
SELECT id, (embedding <=> query_embedding) AS cosine_distance
FROM chunks ORDER BY embedding <=> query_embedding LIMIT 5;

-- Dot Product (for normalized vectors, equivalent to cosine)
SELECT id, (embedding <#> query_embedding) AS dot_product
FROM chunks ORDER BY embedding <#> query_embedding LIMIT 5;

-- Index Types
```

#### 2.4.1 Search Quality Metrics

```python
from typing import List
import numpy as np

class SearchEvaluationMetrics:
    """
    Metrics để đánh giá Search Quality
    Target values cho production system
    """
    
    # NDCG@K - Normalized Discounted Cumulative Gain
    def calculate_ndcg(
        self, 
        relevance_scores: List[float],  # Ground truth relevance
        predicted_scores: List[float],  # Our predicted scores
        k: int = 10
    ) -> float:
        """
        NDCG@K đo lường chất lượng ranking
        Target: > 0.7
        
        Giá trị:
        - 1.0: Perfect ranking
        - 0.9-1.0: Excellent
        - 0.7-0.9: Good
        - 0.5-0.7: Fair
        - < 0.5: Poor
        """
        # Sort by predicted scores
        indices = np.argsort(predicted_scores)[::-1][:k]
        
        # Calculate DCG
        dcg = 0
        for i, idx in enumerate(indices):
            dcg += relevance_scores[idx] / np.log2(i + 2)
        
        # Calculate IDCG (ideal)
        ideal_sorted = sorted(relevance_scores, reverse=True)[:k]
        idcg = sum(
            rel / np.log2(i + 2) 
            for i, rel in enumerate(ideal_sorted)
        )
        
        return dcg / idcg if idcg > 0 else 0
    
    # Precision@K
    def calculate_precision_at_k(
        self, 
        relevant_docs: set,  # Ground truth relevant docs
        predicted_docs: List,  # Our predicted docs
        k: int = 5
    ) -> float:
        """
        Precision@K: Tỷ lệ relevant trong top K
        Target: > 0.8
        
        Precision@5 = |Retrieved Relevant| / 5
        """
        top_k = set(predicted_docs[:k])
        return len(top_k & relevant_docs) / k
    
    # MRR - Mean Reciprocal Rank
    def calculate_mrr(
        self, 
        relevant_docs: set,
        ranked_results: List
    ) -> float:
        """
        MRR: Trung bình nghịch đảo của rank đầu tiên đúng
        Target: > 0.6
        
        Example:
        - Query 1: First relevant at rank 1 → 1/1 = 1.0
        - Query 2: First relevant at rank 3 → 1/3 = 0.33
        - Query 3: No relevant → 0
        - MRR = (1.0 + 0.33 + 0) / 3 = 0.44
        """
        for i, doc in enumerate(ranked_results):
            if doc in relevant_docs:
                return 1 / (i + 1)
        return 0
```

```python
# RAGAS Metrics Implementation
from ragas import evaluate
from ragas.metrics import (
    faithfulness,
    answer_relevancy,
    context_precision,

)

class RAGQualityMetrics:
    """
    RAGAS metrics cho đánh giá RAG system
    """
    
    def evaluate_rag_system(
        self, 
        test_dataset: list
    ) -> dict:
        """
        Evaluate RAG system sử dụng RAGAS metrics
        
        Metrics:
        - Faithfulness: Câu trả lời trung thành với context
        - Answer Relevancy: Câu trả lời liên quan đến câu hỏi
        - Context Precision: Context được retrieve chính xác
        - Context Recall: Context đầy đủ
        """
        
        metrics = [
            faithfulness,        # Target: > 0.8
            answer_relevancy,     # Target: > 0.75
            context_precision,    # Target: > 0.7
            context_recall        # Target: > 0.7
        ]
        
        results = evaluate(
            dataset=test_dataset,
            metrics=metrics
        )
        
        return {
            "faithfulness": results["faithfulness"],
            "answer_relevancy": results["answer_relevancy"],
            "context_precision": results["context_precision"],
            "context_recall": results["context_recall"],
            "overall_score": np.mean([
                results["faithfulness"],
                results["answer_relevancy"],
                results["context_precision"],
                results["context_recall"]
            ])
        }
    
    def calculate_citation_accuracy(
        self,
        generated_answer: str,
        ground_truth_citations: List[str]
    ) -> float:
        """
        Citation Accuracy: Tỷ lệ citations chính xác
        Target: > 90%
        
        Citation phải:
        1. Tồn tại trong context
        2. Được trích dẫn đúng
        3. Không hallucinated
        """
        correct = 0
        total = len(ground_truth_citations)
        
        for citation in ground_truth_citations:
            if citation in generated_answer:
                correct += 1
        
        return correct / total if total > 0 else 0
```

```
│  A:  - Tích hợp tốt với PostgreSQL (giảm infrastructure)       │
```
