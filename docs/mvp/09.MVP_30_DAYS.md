#### Text Extraction Service

```csharp
public interface ITextExtractionService
{
    Task<ExtractedText> ExtractFromPdfAsync(Stream fileStream);
    Task<ExtractedText> ExtractFromTxtAsync(Stream fileStream);
}

public class PdfTextExtractionService : ITextExtractionService
{
    public async Task<ExtractedText> ExtractFromPdfAsync(Stream stream)
    {
        using var reader = new PdfReader(stream);
        using var document = new PdfDocument(reader);
        
        var fullText = new StringBuilder();
        var pageTexts = new List<PageText>();
        
        for (int i = 1; i <= document.GetNumberOfPages(); i++)
        {
            var page = document.GetPage(i);
            var strategy = new SimpleTextExtStrategy();
            var pageText = PdfTextExtractor.GetTextFromPage(page, strategy);
            
            pageTexts.Add(new PageText
            {
                PageNumber = i,
                Text = pageText
            });
            
            fullText.AppendLine(pageText);
        }
        
        return new ExtractedText
        {
            FullText = fullText.ToString(),
            PageTexts = pageTexts,
            PageCount = document.GetNumberOfPages(),
            ExtractedAt = DateTime.UtcNow
        };
    }
}
```

#### Chunking Service

```csharp
public interface IChunkingService
{
    List<DocumentChunk> ChunkText(string text, ChunkingOptions options);
}

public class RecursiveChunkingService : IChunkingService
{
    private const int DefaultChunkSize = 500;
    private const int DefaultOverlap = 50;
    
    public List<DocumentChunk> ChunkText(string text, ChunkingOptions options)
    {
        var chunks = new List<DocumentChunk>();
        var chunkSize = options.ChunkSize ?? DefaultChunkSize;
        var overlap = options.Overlap ?? DefaultOverlap;
        
        // Split by sentences first (better for semantic coherence)
        var sentences = SplitIntoSentences(text);
        
        var currentChunk = new StringBuilder();
        var chunkIndex = 0;
        var startChar = 0;
        
        foreach (var sentence in sentences)
        {
            if (currentChunk.Length + sentence.Length > chunkSize)
            {
                // Save current chunk
                if (currentChunk.Length > 0)
                {
                    chunks.Add(CreateChunk(
                        currentChunk.ToString(),
                        chunkIndex,
                        startChar,
                        options.DocumentId
                    ));
                    
                    chunkIndex++;
                    startChar = Math.Max(0, currentChunk.Length - overlap);
                    
                    // Keep overlap for context
                    var overlapText = currentChunk
                        .ToString()
                        .Substring(Math.Max(0, currentChunk.Length - overlap));
                    
                    currentChunk.Clear();
                    currentChunk.Append(overlapText);
                }
            }
            
            currentChunk.Append(sentence);
        }
        
        // Add remaining chunk
        if (currentChunk.Length > 0)
        {
            chunks.Add(CreateChunk(
                currentChunk.ToString(),
                chunkIndex,
                startChar,
                options.DocumentId
            ));
        }
        
        return chunks;
    }
    
    private List<string> SplitIntoSentences(string text)
    {
        // Simple sentence splitting for Vietnamese
        var pattern = @"[.!?。]+";
        return Regex.Split(text, pattern)
            .Where(s => !string.IsNullOrWhiteSpace(s))
            .Select(s => s.Trim())
            .ToList();
    }
}
```

#### Embedding Service

```csharp
public interface IEmbeddingService
{
    Task<float[]> GenerateEmbeddingAsync(string text);
    Task<List<float[]>> GenerateEmbeddingsBatchAsync(List<string> texts);
}

public class OpenAIEmbeddingService : IEmbeddingService
{
    private readonly HttpClient _httpClient;
    private readonly OpenAISettings _settings;
    private readonly ICacheService _cache;
    
    public async Task<float[]> GenerateEmbeddingAsync(string text)
    {
        // Check cache first
        var cacheKey = $"emb:{text.GetHashCode()}";
        var cached = await _cache.GetAsync<float[]>(cacheKey);
        if (cached != null)
            return cached;
        
        // Call OpenAI API
        var response = await _httpClient.PostAsJsonAsync(
            "https://api.openai.com/v1/embeddings",
            new
            {
                input = text,
                model = "text-embedding-3-small"
            },
            new JsonSerializerOptions
            {
                PropertyNamingPolicy = JsonNamingPolicy.CamelCase
            });
        
        var result = await response.Content
            .ReadFromJsonAsync<OpenAIEmbeddingResponse>();
        
        var embedding = result.Data[0].Embedding;
        
        // Cache the result
        await _cache.SetAsync(cacheKey, embedding, TimeSpan.FromMinutes(5));
        
        return embedding;
    }
    
    public async Task<List<float[]>> GenerateEmbeddingsBatchAsync(
        List<string> texts)
    {
        // Batch API: up to 2048 items per request
        var results = new List<float[]>();
        var batches = texts.Chunk(2048);
        
        foreach (var batch in batches)
        {
            var response = await _httpClient.PostAsJsonAsync(
                "https://api.openai.com/v1/embeddings",
                new
                {
                    input = batch.ToList(),
                    model = "text-embedding-3-small"
                });
            
            var result = await response.Content
                .ReadFromJsonAsync<OpenAIEmbeddingResponse>();
            
            results.AddRange(result.Data
                .OrderBy(d => d.Index)
                .Select(d => d.Embedding));
        }
        
        return results;
    }
}
```

#### Search Service

```csharp
public class SearchService
{
    private readonly AppDbContext _context;
    private readonly IEmbeddingService _embeddingService;
    
    public async Task<SearchResult> SearchAsync(SearchQuery query)
    {
        // Generate embedding for query
        var queryEmbedding = await _embeddingService
            .GenerateEmbeddingAsync(query.QueryText);
        
        // Vector similarity search
        var results = await _context.DocumentChunks
            .FromSqlRaw(@"
                SELECT *, 
                    (content_embedding <=> {0}) AS similarity
                FROM document_chunks
                WHERE is_active = true
                ORDER BY content_embedding <=> {0}
                LIMIT {1}",
                queryEmbedding,
                query.Limit)
            .Include(c => c.Document)
            .ToListAsync();
        
        // Map to response
        return new SearchResult
        {
            Query = query.QueryText,
            Results = results.Select(r => new SearchResultItem
            {
                ChunkId = r.Id,
                DocumentId = r.DocumentId,
                DocumentTitle = r.Document.Title,
                Content = r.Content,
                SimilarityScore = r.Similarity,
                PageNumber = r.PageNumber
            }).ToList(),
            TotalCount = results.Count,
            SearchTimeMs = 0 // Calculate actual time
        };
    }
}
```

#### RAG Pipeline

```csharp
public class RAGPipeline
{
    private readonly ISearchService _searchService;
    private readonly ILLMService _llmService;
    private readonly IAuthorizationService _authService;
    
    private const string SystemPrompt = @"
Bạn là một trợ lý AI chuyên trả lời câu hỏi dựa trên tài liệu được cung cấp.

QUY TẮC NGHIÊM NGẶT:
1. CHỈ trả lời dựa trên thông tin có trong ngữ cảnh được cung cấp
2. NẾU không tìm thấy thông tin liên quan, trả lời: 'Tôi không tìm thấy thông tin này trong tài liệu được cung cấp.'
3. KHÔNG được bịa đặt hoặc suy luận thông tin không có trong ngữ cảnh
4. LUÔN trích dẫn nguồn khi trả lời

Định dạng trả lời:
- Câu trả lời: [nội dung trả lời dựa trên ngữ cảnh]
- Nguồn: [trích dẫn tài liệu, trang, đoạn]
";
    
    public async Task<RAGResponse> GenerateResponseAsync(
        string question,
        Guid userId,
        List<Guid>? documentIds = null)
    {
        // Step 1: Retrieve relevant chunks
        var searchResults = await _searchService.SearchAsync(
            new SearchQuery 
            { 
                QueryText = question,
                Limit = 5,
                DocumentIds = documentIds
            });
        
        // Filter by authorization
        var authorizedResults = new List<SearchResultItem>();
        foreach (var result in searchResults.Results)
        {
            if (await _authService.CanAccessDocumentAsync(
                userId, result.DocumentId))
            {
                authorizedResults.Add(result);
            }
        }
        
        // Step 2: Check if we have relevant context
        if (!authorizedResults.Any())
        {
            return new RAGResponse
            {
                Answer = "Tôi không tìm thấy thông tin liên quan trong tài liệu được cung cấp.",
                Sources = new List<Source>(),
                Confidence = ConfidenceLevel.Low
            };
        }
        
        // Step 3: Build context from retrieved chunks
        var context = BuildContext(authorizedResults);
        
        // Step 4: Build prompt
        var prompt = $@"
Ngữ cảnh:
{context}

Câu hỏi: {question}

Hãy trả lời câu hỏi dựa trên ngữ cảnh được cung cấp.
";
        
        // Step 5: Generate response
        var response = await _llmService.GenerateChatCompletionAsync(
            new ChatRequest
            {
                SystemPrompt = SystemPrompt,
                UserMessage = prompt
            });
        
        // Step 6: Extract citations
        var citations = ExtractCitations(authorizedResults);
        
        return new RAGResponse
        {
            Answer = response.Content,
            Sources = citations,
            Confidence = CalculateConfidence(searchResults.Results)
        };
    }
    
    private string BuildContext(List<SearchResultItem> results)
    {
        var context = new StringBuilder();
        
        foreach (var result in results)
        {
            context.AppendLine($"--- Tài liệu: {result.DocumentTitle} ---");
            if (result.PageNumber.HasValue)
                context.AppendLine($"Trang: {result.PageNumber}");
            context.AppendLine(result.Content);
            context.AppendLine();
        }
        
        return context.ToString();
    }
    
    private List<Source> ExtractCitations(List<SearchResultItem> results)
    {
        return results.Select(r => new Source
        {
            DocumentId = r.DocumentId,
            DocumentTitle = r.DocumentTitle,
            ChunkId = r.ChunkId,
            PageNumber = r.PageNumber,
            SimilarityScore = r.SimilarityScore
        }).ToList();
    }
    
    private ConfidenceLevel CalculateConfidence(List<SearchResultItem> results)
    {
        if (!results.Any())
            return ConfidenceLevel.Low;
        
        var avgSimilarity = results.Average(r => r.SimilarityScore);
        
        if (avgSimilarity > 0.8)
            return ConfidenceLevel.High;
        if (avgSimilarity > 0.6)
            return ConfidenceLevel.Medium;
        return ConfidenceLevel.Low;
    }
}
```
