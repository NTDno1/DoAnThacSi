## MODULE 1: AUTH MODULE

### 1.1 Mục Đích

Module xác thực và phân quyền người dùng (Authentication & Authorization) là nền tảng bảo mật cho toàn bộ hệ thống. Module này đảm bảo:

- **Xác thực (Authentication)**: Xác minh danh tính người dùng qua credentials (username/password)
- **Ủy quyền (Authorization)**: Kiểm soát quyền truy cập tài nguyên dựa trên vai trò (RBAC)
- **Quản lý phiên (Session Management)**: Theo dõi và quản lý phiên đăng nhập
- **Bảo mật**: Mã hóa mật khẩu, bảo vệ against common attacks

#### 1.3.1 Đăng Ký Người Dùng (User Registration)

```csharp
// Services/AuthService.cs
public class AuthService : IAuthService
{
    private readonly IUserRepository _userRepository;
    private readonly IEmailService _emailService;
    private readonly IUnitOfWork _unitOfWork;
    private readonly IOptions<JwtSettings> _jwtSettings;
    
    public async Task<AuthResult> RegisterAsync(RegisterRequest request)
    {
        // 1. Validate request
        if (!await ValidateRegistrationRequest(request))
            return AuthResult.Failed("Thông tin đăng ký không hợp lệ");
        
        // 2. Check existing user
        if (await _userRepository.ExistsByEmailAsync(request.Email))
            return AuthResult.Failed("Email đã được sử dụng");
        
        if (await _userRepository.ExistsByUsernameAsync(request.Username))
            return AuthResult.Failed("Tên đăng nhập đã tồn tại");
        
        // 3. Create user entity
        var user = new User
        {
            Id = Guid.NewGuid(),
            Username = request.Username,
            Email = request.Email.ToLowerInvariant(),
            FullName = request.FullName,
            IsActive = false, // Chờ xác nhận email
            EmailConfirmed = false
        };
        
        // 4. Hash password với salt
        var (hash, salt) = PasswordHasher.Hash(request.Password);
        user.PasswordHash = hash;
        user.PasswordSalt = salt;
        
        // 5. Assign default role (Viewer)
        var viewerRole = await _userRepository.GetRoleByNameAsync("Viewer");
        if (viewerRole != null)
        {
            user.UserRoles.Add(new UserRole
            {
                UserId = user.Id,
                RoleId = viewerRole.Id,
                AssignedAt = DateTime.UtcNow
            });
        }
        
        // 6. Generate email confirmation token
        var confirmationToken = GenerateEmailConfirmationToken(user);
        
        // 7. Save to database
        await _userRepository.AddAsync(user);
        await _unitOfWork.SaveChangesAsync();
        
        // 8. Send confirmation email (async, non-blocking)
        _ = _emailService.SendEmailConfirmationAsync(user.Email, confirmationToken);
        
        return AuthResult.Success(new AuthResponse
        {
            UserId = user.Id,
            Message = "Đăng ký thành công. Vui lòng xác nhận email để kích hoạt tài khoản."
        });
    }
    
    private (string Hash, string Salt) Hash(string password)
    {
        using var rng = RandomNumberGenerator.Create();
        var saltBytes = new byte[128 / 8];
        rng.GetBytes(saltBytes);
        
        var salt = Convert.ToBase64String(saltBytes);
        var pepper = _jwtSettings.Value.PasswordPepper; // Server-side secret
        
        using var pbkdf2 = new Rfc2898DeriveBytes(
            password + pepper,
            saltBytes,
            100000, // iterations
            HashAlgorithmName.SHA512);
        
        var hash = Convert.ToBase64String(pbkdf2.GetBytes(256 / 8));
        return (hash, salt);
    }
}
```

#### 1.3.2 Đăng Nhập (Login)

```csharp
public async Task<AuthResult> LoginAsync(LoginRequest request)
{
    // 1. Rate limiting check
    var clientIp = GetClientIpAddress();
    if (await _rateLimiter.IsBlockedAsync(clientIp))
        return AuthResult.Failed("Tài khoản đã bị khóa tạm thời. Vui lòng thử lại sau 15 phút.");
    
    // 2. Find user by username or email
    var user = await _userRepository.GetByUsernameOrEmailAsync(request.UsernameOrEmail);
    if (user == null)
    {
        await _rateLimiter.RecordFailedAttemptAsync(clientIp);
        return AuthResult.Failed("Tên đăng nhập hoặc mật khẩu không đúng");
    }
    
    // 3. Check account lockout
    if (user.LockoutEnd.HasValue && user.LockoutEnd > DateTime.UtcNow)
    {
        var remainingTime = (user.LockoutEnd.Value - DateTime.UtcNow).Minutes;
        return AuthResult.Failed($"Tài khoản bị khóa. Vui lòng thử lại sau {remainingTime} phút.");
    }
    
    // 4. Verify password
    if (!PasswordHasher.Verify(request.Password, user.PasswordHash, user.PasswordSalt))
    {
        user.FailedLoginAttempts++;
        
        if (user.FailedLoginAttempts >= 5)
        {
            user.LockoutEnd = DateTime.UtcNow.AddMinutes(15);
        }
        
        await _unitOfWork.SaveChangesAsync();
        await _rateLimiter.RecordFailedAttemptAsync(clientIp);
        
        return AuthResult.Failed("Tên đăng nhập hoặc mật khẩu không đúng");
    }
    
    // 5. Check account status
    if (!user.IsActive)
        return AuthResult.Failed("Tài khoản chưa được kích hoạt. Vui lòng xác nhận email.");
    
    // 6. Reset failed attempts on successful login
    user.FailedLoginAttempts = 0;
    user.LockoutEnd = null;
    user.LastLoginAt = DateTime.UtcNow;
    
    // 7. Generate tokens
    var accessToken = GenerateJwtToken(user);
    var refreshToken = await GenerateRefreshTokenAsync(user, request.DeviceInfo, clientIp);
    
    // 8. Log successful login
    await _auditLog.LogAsync(new AuditLog
    {
        UserId = user.Id,
        Action = "USER_LOGIN",
        IpAddress = clientIp,
        EntityType = "User"
    });
    
    await _unitOfWork.SaveChangesAsync();
    
    return AuthResult.Success(new AuthResponse
    {
        AccessToken = accessToken,
        RefreshToken = refreshToken.Token,
        ExpiresIn = _jwtSettings.Value.AccessTokenExpiration,
        TokenType = "Bearer",
        User = MapToUserDto(user)
    });
}

private string GenerateJwtToken(User user)
{
    var claims = new List<Claim>
    {
        new(JwtRegisteredClaimNames.Sub, user.Id.ToString()),
        new(JwtRegisteredClaimNames.Email, user.Email),
        new(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString()),
        new("username", user.Username),
        new("fullname", user.FullName)
    };
    
    // Add role claims
    foreach (var userRole in user.UserRoles)
    {
        claims.Add(new Claim(ClaimTypes.Role, userRole.Role.Name));
    }
    
    // Add permission claims
    var permissions = user.UserRoles
        .SelectMany(ur => ur.Role.RolePermissions)
        .Select(rp => rp.Permission.Name)
        .Distinct();
    
    foreach (var permission in permissions)
    {
        claims.Add(new Claim("permission", permission));
    }
    
    var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_jwtSettings.Value.SecretKey));
    var credentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha512Signature);
    
    var token = new JwtSecurityToken(
        issuer: _jwtSettings.Value.Issuer,
        audience: _jwtSettings.Value.Audience,
        claims: claims,
        expires: DateTime.UtcNow.AddMinutes(_jwtSettings.Value.AccessTokenExpiration),
        signingCredentials: credentials
    );
    
    return new JwtSecurityTokenHandler().WriteToken(token);
}
```

#### 1.3.3 Refresh Token

```csharp
public async Task<AuthResult> RefreshTokenAsync(string refreshToken)
{
    // 1. Find refresh token in database
    var storedToken = await _refreshTokenRepository.GetByTokenAsync(refreshToken);
    
    if (storedToken == null)
        return AuthResult.Failed("Refresh token không hợp lệ");
    
    if (storedToken.IsExpired)
        return AuthResult.Failed("Refresh token đã hết hạn");
    
    if (storedToken.IsRevoked)
        return AuthResult.Failed("Refresh token đã bị thu hồi");
    
    // 2. Mark old token as used
    storedToken.UsedAt = DateTime.UtcNow;
    
    // 3. Get user
    var user = await _userRepository.GetByIdWithRolesAsync(storedToken.UserId);
    if (user == null || !user.IsActive)
        return AuthResult.Failed("Người dùng không tồn tại hoặc không hoạt động");
    
    // 4. Generate new tokens (rotation)
    var newAccessToken = GenerateJwtToken(user);
    var newRefreshToken = await GenerateRefreshTokenAsync(user, storedToken.DeviceInfo, storedToken.IpAddress);
    
    // 5. Revoke old refresh token family to prevent reuse
    await RevokeRefreshTokenFamilyAsync(storedToken.UserId, storedToken.Id);
    
    await _unitOfWork.SaveChangesAsync();
    
    return AuthResult.Success(new AuthResponse
    {
        AccessToken = newAccessToken,
        RefreshToken = newRefreshToken.Token,
        ExpiresIn = _jwtSettings.Value.AccessTokenExpiration,
        TokenType = "Bearer"
    });
}

private async Task RevokeRefreshTokenFamilyAsync(Guid userId, Guid usedTokenId)
{
    // Revoke all refresh tokens for this user that were issued before the used token
    // This prevents token theft via "refresh token rotation attack"
    var tokensToRevoke = await _refreshTokenRepository
        .GetActiveTokensByUserIdAsync(userId);
    
    foreach (var token in tokensToRevoke.Where(t => t.Id != usedTokenId && t.CreatedAt < DateTime.UtcNow.AddHours(-24)))
    {
        token.RevokedAt = DateTime.UtcNow;
    }
}
```

```csharp
// Authorization/Requirements/PermissionRequirement.cs
public class PermissionRequirement : IAuthorizationRequirement
{
    public string Permission { get; }
    
    public PermissionRequirement(string permission)
    {
        Permission = permission;
    }
}

public class PermissionHandler : AuthorizationHandler<PermissionRequirement>
{
    private readonly IHttpContextAccessor _httpContextAccessor;
    
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        PermissionRequirement requirement)
    {
        var permissions = context.User.Claims
            .Where(c => c.Type == "permission")
            .Select(c => c.Value)
            .ToList();
        
        if (permissions.Contains(requirement.Permission) || permissions.Contains("*"))
        {
            context.Succeed(requirement);
        }
        
        return Task.CompletedTask;
    }
}
```

### 1.6 Edge Cases & Error Handling

```csharp
public class AuthException : Exception
{
    public AuthErrorCode ErrorCode { get; }
    
    public AuthException(AuthErrorCode errorCode, string message) : base(message)
    {
        ErrorCode = errorCode;
    }
}

public enum AuthErrorCode
{
    InvalidCredentials,
    AccountLocked,
    AccountDisabled,
    EmailNotConfirmed,
    RefreshTokenExpired,
    RefreshTokenRevoked,
    InvalidRefreshToken,
    PasswordRequirementsNotMet,
    EmailAlreadyExists,
    UsernameAlreadyExists,
    RateLimitExceeded
}

// Global exception handler
public class AuthExceptionFilter : IExceptionFilter
{
    public void OnException(ExceptionContext context)
    {
        if (context.Exception is AuthException authEx)
        {
            var (statusCode, errorResponse) = authEx.ErrorCode switch
            {
                AuthErrorCode.InvalidCredentials => 
                    (StatusCodes.Status401Unauthorized, new ErrorResponse { Code = "AUTH_001", Message = authEx.Message }),
                AuthErrorCode.AccountLocked => 
                    (StatusCodes.Status423Locked, new ErrorResponse { Code = "AUTH_002", Message = authEx.Message }),
                AuthErrorCode.RateLimitExceeded => 
                    (StatusCodes.Status429TooManyRequests, new ErrorResponse { Code = "AUTH_003", Message = authEx.Message }),
                _ => 
                    (StatusCodes.Status400BadRequest, new ErrorResponse { Code = "AUTH_999", Message = authEx.Message })
            };
            
            context.Result = new ObjectResult(errorResponse) { StatusCode = statusCode };
            context.ExceptionHandled = true;
        }
    }
}
```

1. **Concurrent Login**: Cho phép nhiều phiên đăng nhập cùng lúc nhưng giới hạn số lượng
2. **Token Theft**: Phát hiện và revoke token family khi phát hiện sử dụng bất thường
3. **Password Policy**: Minimum 8 ký tự, chữ hoa, chữ thường, số, ký tự đặc biệt
4. **Email Verification**: Token expires trong 24 giờ, resend có cooldown 60 giây
5. **Session Timeout**: Sliding expiration, tự động extend khi có activity
6. **MFA (Multi-Factor Authentication)**: Tùy chọn cho tài khoản admin (OTP via TOTP)

### 1.7 Dependencies

- **PostgreSQL**: Lưu trữ Users, Roles, Permissions, RefreshTokens, AuditLogs
- **Redis**: Cache user sessions, rate limiting, blacklist tokens
- **Email Service**: SendGrid/AWS SES cho confirmation emails
- **Password Hasher**: BCrypt.Net-Next hoặc PBKDF2

```csharp
// Services/DocumentService.cs
public class DocumentService : IDocumentService
{
    private readonly IDocumentRepository _documentRepository;
    private readonly IMinioService _minioService;
    private readonly IOcrService _ocrService;
    private readonly IChunkingService _chunkingService;
    private readonly IEmbeddingService _embeddingService;
    private readonly IBackgroundJobClient _jobClient;
    private readonly IUnitOfWork _unitOfWork;
    private readonly ILogger<DocumentService> _logger;
    
    // Supported file formats
    private static readonly Dictionary<string, string[]> SupportedFormats = new()
    {
        { "application/pdf", new[] { ".pdf" } },
        { "application/vnd.openxmlformats-officedocument.wordprocessingml.document", new[] { ".docx" } },
        { "application/msword", new[] { ".doc" } },
        { "text/plain", new[] { ".txt" } },
        { "application/rtf", new[] { ".rtf" } },
        { "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", new[] { ".xlsx" } },
        { "application/vnd.ms-excel", new[] { ".xls" } },
        { "application/vnd.openxmlformats-officedocument.presentationml.presentation", new[] { ".pptx" } },
        { "application/vnd.ms-powerpoint", new[] { ".ppt" } },
        { "image/png", new[] { ".png" } },
        { "image/jpeg", new[] { ".jpg", ".jpeg" } },
        { "image/tiff", new[] { ".tiff", ".tif" } },
        { "image/bmp", new[] { ".bmp" } }
    };
    
    public async Task<DocumentUploadResult> UploadAsync(UploadDocumentRequest request, Guid userId)
    {
        // 1. Validate file
        var validationResult = await ValidateFileAsync(request.File);
        if (!validationResult.IsValid)
            return DocumentUploadResult.Failed(validationResult.Error!);
        
        // 2. Check storage quota
        var userQuota = await _userRepository.GetStorageQuotaAsync(userId);
        if (userQuota.UsedBytes + request.File.Length > userQuota.MaxBytes)
            return DocumentUploadResult.Failed($"Vượt quá dung lượng lưu trữ. Tối đa {userQuota.MaxGB}GB");
        
        // 3. Generate storage path
        var storagePath = GenerateStoragePath(request.File.FileName, userId);
        
        // 4. Upload to MinIO
        using var stream = request.File.OpenReadStream();
        await _minioService.UploadAsync(storagePath, stream, request.File.ContentType);
        
        // 5. Create document record
        var document = new Document
        {
            Id = Guid.NewGuid(),
            Title = request.Title ?? Path.GetFileNameWithoutExtension(request.File.FileName),
            Description = request.Description,
            FileName = request.File.FileName,
            FileExtension = Path.GetExtension(request.File.FileName).ToLowerInvariant(),
            MimeType = request.File.ContentType,
            FileSizeBytes = request.File.Length,
            StoragePath = storagePath,
            Status = DocumentStatus.Uploaded,
            UploadedByUserId = userId,
            AccessLevel = request.AccessLevel,
            CategoryId = request.CategoryId,
            UploadedAt = DateTime.UtcNow
        };
        
        // 6. Handle OCR for images/scanned PDFs
        if (IsOcrRequired(request.File.ContentType))
        {
            document.Status = DocumentStatus.Pending;
            document.ExtractedText = null;
        }
        else
        {
            document.Status = DocumentStatus.Uploaded;
        }
        
        await _documentRepository.AddAsync(document);
        await _unitOfWork.SaveChangesAsync();
        
        // 7. Queue background job for processing
        if (document.Status == DocumentStatus.Pending)
        {
            _jobClient.Enqueue(() => ProcessDocumentAsync(document.Id));
        }
        
        return DocumentUploadResult.Success(document);
    }
    
    public async Task ProcessDocumentAsync(Guid documentId)
    {
        var document = await _documentRepository.GetByIdAsync(documentId);
        if (document == null) return;
        
        try
        {
            document.Status = DocumentStatus.Processing;
            await _unitOfWork.SaveChangesAsync();
            
            // Step 1: OCR / Text Extraction
            string extractedText;
            if (IsImageFile(document.FileExtension))
            {
                var imageStream = await _minioService.GetObjectAsync(document.StoragePath);
                var ocrResult = await _ocrService.ExtractTextAsync(imageStream, document.FileExtension);
                extractedText = ocrResult.Text;
                document.TextLanguage = ocrResult.Language;
            }
            else if (document.FileExtension == ".pdf")
            {
                var pdfStream = await _minioService.GetObjectAsync(document.StoragePath);
                var ocrResult = await _ocrService.ExtractTextFromPdfAsync(pdfStream);
                extractedText = ocrResult.Text;
                document.TextLanguage = ocrResult.Language;
                document.PageCount = ocrResult.PageCount;
            }
            else
            {
                // Plain text or Office documents - extract directly
                extractedText = await ExtractTextFromOfficeDocAsync(document.StoragePath);
                document.TextLanguage = DetectLanguage(extractedText);
            }
            
            document.ExtractedText = extractedText;
            document.WordCount = WordCount(extractedText);
            
            // Step 2: Chunking
            var chunks = await _chunkingService.ChunkTextAsync(extractedText, document.Id);
            
            // Step 3: Generate embeddings
            await _embeddingService.EmbedChunksAsync(chunks);
            
            document.Status = DocumentStatus.Processed;
            document.ProcessedAt = DateTime.UtcNow;
            
            _logger.LogInformation("Document {DocumentId} processed successfully. {ChunkCount} chunks created.",
                documentId, chunks.Count);
        }
        catch (Exception ex)
        {
            document.Status = DocumentStatus.Failed;
            document.ProcessingError = ex.Message;
            _logger.LogError(ex, "Failed to process document {DocumentId}", documentId);
        }
        
        await _unitOfWork.SaveChangesAsync();
    }
    
    private static bool IsOcrRequired(string mimeType)
    {
        return mimeType.StartsWith("image/") || 
               mimeType == "application/pdf";
    }
    
    private static string GenerateStoragePath(string fileName, Guid userId)
    {
        var date = DateTime.UtcNow;
        var sanitizedName = SanitizeFileName(fileName);
        return $"documents/{userId}/{date:yyyy/MM/dd}/{Guid.NewGuid()}_{sanitizedName}";
    }
}
```

```csharp
public async Task<string> GetDownloadUrlAsync(Guid documentId, Guid userId)
{
    var document = await _documentRepository.GetByIdAsync(documentId);
    if (document == null)
        throw new DocumentNotFoundException(documentId);
    
    // Check access permission
    if (!await HasAccessAsync(document, userId))
        throw new UnauthorizedAccessException("Bạn không có quyền tải tài liệu này");
    
    // Log access
    await _documentRepository.LogAccessAsync(documentId, userId, DocumentAccessType.Download);
    
    // Generate presigned URL (expires in 1 hour)
    var url = await _minioService.GetPresignedUrlAsync(
        document.StoragePath,
        TimeSpan.FromHours(1));
    
    return url;
}

private async Task<bool> HasAccessAsync(Document document, Guid userId)
{
    if (document.IsPublic || document.AccessLevel == DocumentAccessLevel.Public)
        return true;
    
    if (document.UploadedByUserId == userId)
        return true;
    
    var user = await _userRepository.GetByIdAsync(userId);
    if (user?.UserRoles.Any(r => r.Role.Name == "Admin") == true)
        return true;
    
    // Check organization access
    if (document.AccessLevel == DocumentAccessLevel.Organization)
    {
        var uploaderOrg = await _userRepository.GetOrganizationAsync(document.UploadedByUserId);
        var userOrg = await _userRepository.GetOrganizationAsync(userId);
        return uploaderOrg == userOrg;
    }
    
    return false;
}
```

```csharp
    if (request.File.Length > 100 * 1024 * 1024)
        return DocumentUploadResult.Failed("File quá lớn. Tối đa 100MB");
    
    // 2. Corrupted file detection
    try
    {
        using var reader = new BinaryReader(request.File.OpenReadStream());

        if (!IsValidFileFormat(header, request.File.ContentType))
            return DocumentUploadResult.Failed("File không hợp lệ hoặc bị hỏng");

    catch (Exception ex)
    {
        _logger.LogWarning(ex, "Could not validate file header");
    }
    
    // 3. Duplicate detection (same hash)
    var fileHash = await ComputeFileHashAsync(request.File);
    var existingDoc = await _documentRepository.FindByHashAsync(fileHash);
    if (existingDoc != null)
        return DocumentUploadResult.Failed($"File đã tồn tại: {existingDoc.Title}");
    
    // 4. Concurrent upload protection
    var lockKey = $"upload:{userId}:{request.File.FileName}";
    if (!await _distributedLock.AcquireAsync(lockKey, TimeSpan.FromMinutes(5)))
        return DocumentUploadResult.Failed("Đang có upload khác đang thực hiện");
    
    try { ... }
    finally { await _distributedLock.ReleaseAsync(lockKey); }
}
```

- Layout analysis (headers, paragraphs, footnotes)

```csharp
// Services/OcrService.cs
public class OcrService : IOcrService
{
    private readonly PaddleOcrApi _paddleOcr;
    private readonly IImageProcessor _imageProcessor;
    private readonly ILanguageDetector _languageDetector;
    private readonly ILogger<OcrService> _logger;
    
    public async Task<OcrResult> ExtractTextAsync(Stream imageStream, string fileExtension)
    {
        var result = new OcrResult
        {
            Id = Guid.NewGuid(),
            ProcessingTime = new ProcessingTime { StartedAt = DateTime.UtcNow }
        };
        
        try
        {
            // Step 1: Preprocessing
            var preprocessedImage = await PreprocessImageAsync(imageStream, fileExtension);
            result.ProcessingTime.PreprocessingDuration = DateTime.UtcNow - result.ProcessingTime.StartedAt;
            
            // Step 2: Layout Analysis
            var layout = await AnalyzeLayoutAsync(preprocessedImage);
            
            // Step 3: OCR per block type
            foreach (var block in layout.Blocks)
            {
                var blockResult = await ProcessBlockAsync(preprocessedImage, block);
                result.TextBlocks.Add(blockResult);
            }
            
            // Step 4: Table Detection & Extraction
            var tables = await ExtractTablesAsync(preprocessedImage, layout.TableRegions);
            result.Tables = tables;
            
            // Step 5: Post-processing
            result.FullText = await PostProcessTextAsync(result.TextBlocks, result.Tables);
            result.ProcessingTime.OcrDuration = DateTime.UtcNow - result.ProcessingTime.StartedAt - result.ProcessingTime.PreprocessingDuration;
            
            // Step 6: Language Detection
            result.DetectedLanguage = await _languageDetector.DetectAsync(result.FullText);
            
            // Step 7: Quality Assessment
            result.OverallConfidence = CalculateOverallConfidence(result.TextBlocks);
            result.RequiresManualReview = ShouldRequireManualReview(result);
            
            result.ProcessingTime.CompletedAt = DateTime.UtcNow;
            result.TotalCharacters = result.FullText.Length;
            
            _logger.LogInformation(
                "OCR completed. Language: {Lang}, Confidence: {Conf:F2}%, Characters: {CharCount}",
                result.DetectedLanguage, result.OverallConfidence * 100, result.TotalCharacters);
            
            return result;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "OCR failed");
            throw new OcrProcessingException("Failed to extract text from image", ex);
        }
    }
    
    private async Task<byte[]> PreprocessImageAsync(Stream imageStream, string fileExtension)
    {
        using var inputStream = new MemoryStream();
        await imageStream.CopyToAsync(inputStream);
        inputStream.Position = 0;
        
        var image = Image.Load<Rgba32>(inputStream);
        
        // 1. Convert to grayscale if needed
        if (image.channels != 1)
        {
            image.Mutate(x => x.Grayscale());
        }
        
        // 2. Denoise (Gaussian blur + threshold)
        image.Mutate(x => x.GaussianBlur(0.5));
        
        // 3. Contrast enhancement
        image.Mutate(x => x.Brightness(1.1));
        image.Mutate(x => x.Contrast(1.2));
        
        // 4.deskew (auto-rotate if needed)
        var rotation = await DetectSkewAngleAsync(image);
        if (Math.Abs(rotation) > 0.5)
        {
            image.Mutate(x => x.Rotate(-rotation));
        }
        
        // 5. Resize if too large (maintain aspect ratio)
        const int maxDimension = 4096;
        if (image.Width > maxDimension || image.Height > maxDimension)
        {
            var ratio = Math.Min(maxDimension / image.Width, maxDimension / image.Height);
            image.Mutate(x => x.Resize((int)(image.Width * ratio), (int)(image.Height * ratio)));
        }
        
        // 6. Convert to PNG for processing
        using var outputStream = new MemoryStream();
        image.SaveAsPng(outputStream);
        return outputStream.ToArray();
    }
    
    private async Task<LayoutAnalysis> AnalyzeLayoutAsync(byte[] imageData)
    {
        // Use PaddleOCR's layout detection model
        var layoutResult = await _paddleOcr.DetectLayoutAsync(imageData);
        
        return new LayoutAnalysis
        {
            Blocks = layoutResult.Blocks.Select(b => new LayoutBlock
            {
                Type = MapLayoutType(b.Type),
                BoundingBox = b.BoundingBox,
                PageNumber = b.PageNumber
            }).ToList(),
            TableRegions = layoutResult.Tables.Select(t => t.Region).ToList()
        };
    }
    
    private async Task<TextBlock> ProcessBlockAsync(byte[] image, LayoutBlock block)
    {
        // Crop image to block region
        var croppedImage = CropImage(image, block.BoundingBox);
        
        // Apply block-specific preprocessing
        croppedImage = block.Type switch
        {
            TextBlockType.Header => EnhanceForSmallText(croppedImage),
            TextBlockType.Table => EnhanceForTable(croppedImage),
            TextBlockType.Signature => EnhanceForSignature(croppedImage),
            _ => croppedImage
        };
        
        // Run OCR
        var ocrResult = await _paddleOcr.RecognizeTextAsync(croppedImage, 
            new OcrOptions 
            { 
                Language = "en,vi", // Multi-language
                UseAngleCls = true, // Text rotation detection
                Use_gpu = true // GPU acceleration if available
            });
        
        return new TextBlock
        {
            BlockIndex = block.Index,
            PageNumber = block.PageNumber,
            BoundingBox = block.BoundingBox,
            Text = ocrResult.Text,
            Confidence = ocrResult.Confidence,
            BlockType = block.Type
        };
    }
    
    private double CalculateOverallConfidence(List<TextBlock> blocks)
    {
        if (!blocks.Any()) return 0;
        
        // Weighted average - higher weight for longer blocks
        var weightedSum = blocks.Sum(b => b.Confidence * b.Text.Length);
        var totalLength = blocks.Sum(b => b.Text.Length);
        
        return totalLength > 0 ? weightedSum / totalLength : 0;
    }
    
    private bool ShouldRequireManualReview(OcrResult result)
    {
        // Flag for manual review if:
        // 1. Overall confidence < 70%
        if (result.OverallConfidence < 0.7) return true;
        
        // 2. Very short text extracted (might be image without text)
        if (result.TotalCharacters < 50 && result.TextBlocks.Count < 3) return true;
        
        // 3. High percentage of low-confidence blocks
        var lowConfidenceBlocks = result.TextBlocks.Count(b => b.Confidence < 0.5);
        if (lowConfidenceBlocks > result.TextBlocks.Count * 0.3) return true;
        
        return false;
    }
}
```

```csharp
public async Task<OcrResult> ExtractTextFromPdfAsync(Stream pdfStream)
{
    var result = new OcrResult
    {
        Id = Guid.NewGuid(),
        ProcessingTime = new ProcessingTime { StartedAt = DateTime.UtcNow }
    };
    
    using var document = PdfDocument.Load(pdfStream);
    result.PageCount = document.NumberOfPages;
    
    var allText = new StringBuilder();
    
    for (int pageNum = 1; pageNum <= document.NumberOfPages; pageNum++)
    {
        var page = document.GetPage(pageNum);
        
        // Check if page has embedded text
        var embeddedText = page.ExtractText();
        var hasText = !string.IsNullOrWhiteSpace(embeddedText) && embeddedText.Length > 10;
        
        if (hasText)
        {
            // Page has readable text
            allText.AppendLine($"--- Page {pageNum} ---");
            allText.AppendLine(embeddedText);
            
            result.TextBlocks.Add(new TextBlock
            {
                BlockIndex = pageNum,
                PageNumber = pageNum,
                Text = embeddedText,
                Confidence = 1.0, // Embedded text is 100% confident
                BlockType = TextBlockType.Paragraph
            });
        }
        else
        {
            // Page is likely scanned - need OCR
            var images = page.GetImages();
            foreach (var image in images)
            {
                using var imageStream = new MemoryStream(image.ImageBytes);
                var ocrResult = await ExtractTextAsync(imageStream, image.Format);
                
                allText.AppendLine($"--- OCR Page {pageNum} ---");
                allText.AppendLine(ocrResult.FullText);
                
                result.TextBlocks.AddRange(ocrResult.TextBlocks.Select(b =>
                {
                    b.PageNumber = pageNum;
                    return b;
                }));
            }
            
            // If no images found, try to render page as image
            if (!images.Any())
            {
                var renderedImage = page.Render(300); // 300 DPI
                var ocrResult = await ExtractTextAsync(
                    new MemoryStream(renderedImage), ".png");
                
                allText.AppendLine($"--- OCR Page {pageNum} ---");
                allText.AppendLine(ocrResult.FullText);
                
                result.TextBlocks.AddRange(ocrResult.TextBlocks.Select(b =>
                {
                    b.PageNumber = pageNum;
                    return b;
                }));
            }
        }
    }
    
    result.FullText = allText.ToString();
    result.DetectedLanguage = await _languageDetector.DetectAsync(result.FullText);
    result.TotalCharacters = result.FullText.Length;
    result.ProcessingTime.CompletedAt = DateTime.UtcNow;
    
    return result;
}
```

```csharp
public async Task<OcrResult> SafeExtractAsync(Stream stream, string extension)
{
    try
    {
        // 1. Check image dimensions
        using var ms = new MemoryStream();
        await stream.CopyToAsync(ms);
        ms.Position = 0;
        
        var imageInfo = await Image.IdentifyAsync(ms);
        if (imageInfo.Width < 100 || imageInfo.Height < 100)
            throw new OcrProcessingException(OcrErrorCode.ImageTooSmall, "Image too small for OCR");
        
        if (imageInfo.Width > 10000 || imageInfo.Height > 10000)
            throw new OcrProcessingException(OcrErrorCode.ImageTooLarge, "Image too large");
        
        ms.Position = 0;
        
        // 2. Validate image quality
        var quality = await ValidateImageQualityAsync(ms);
        if (quality.Score < 0.3)
        {
            _logger.LogWarning("Low quality image detected. Confidence may be reduced.");
        }
        
        // 3. Process with timeout
        using var cts = new CancellationTokenSource(TimeSpan.FromMinutes(5));
        return await ExtractTextAsync(ms, extension).WithCancellation(cts.Token);
    }
    catch (OcrProcessingException)
    {
        throw; // Re-throw known exceptions
    }
    catch (OperationCanceledException)
    {
        throw new OcrProcessingException(OcrErrorCode.ProcessingTimeout, 
            "OCR processing timed out after 5 minutes");
    }
    catch (Exception ex)
    {
        throw new OcrProcessingException(OcrErrorCode.EngineFailure, 
            "OCR engine failed", ex);
    }
}
```

**Các Edge Cases cần xử lý:**

1. **Low-resolution images**: Skip OCR if DPI < 72
2. **Rotated images**: Auto-detect and correct rotation
3. **Multi-column layouts**: Preserve column order
4. **Handwriting**: Flag for manual review (low confidence)
5. **Mixed languages**: Detect and handle multiple languages
6. **Tables with merged cells**: Proper span handling
7. **Complex layouts**: Headers spanning multiple columns
8. **QR codes/barcodes**: Ignore or extract as metadata

```csharp
    public PaddleOcrService(IOptions<OcrSettings> settings)
    {

        var recModelPath = settings.Value.RecognitionModelPath;
        var clsModelPath = settings.Value.ClassificationModelPath;

    public async Task<OcrBlockResult> RecognizeTextAsync(
        byte[] imageData, 
        OcrOptions options)
    {
        var results = new List<(string Text, double Confidence, BoundingBox Box)>();
        
        // 1. Detect text regions
        var detectedRegions = await _detector.DetectAsync(imageData);
        
        // 2. Classify text direction (0°, 90°, 180°, 270°)
        foreach (var region in detectedRegions)
        {
            var angle = await _classifier.ClassifyAsync(CropImage(imageData, region));
            var rotatedRegion = RotateBoundingBox(region, angle);
            
            // 3. Recognize text
            var (text, confidence) = await _recognizer.RecognizeAsync(
                CropImage(imageData, rotatedRegion),
                language: options.Language,
                useGpu: options.Use_gpu
            );
            
            results.Add((text, confidence, region));
        }
        
        return new OcrBlockResult
        {
            Blocks = results,
            OverallConfidence = results.Average(r => r.Confidence)
        };
    }
}
```

```csharp
// Batch processing with parallelism control
public async Task<List<OcrResult>> BatchExtractAsync(
    IEnumerable<(Stream Stream, string Extension)> items,
    int batchSize = 10,
    int maxConcurrency = 3)
{
    var results = new ConcurrentBag<OcrResult>();
    var semaphore = new SemaphoreSlim(maxConcurrency);
    
    var batches = items.Chunk(batchSize);
    
    await foreach (var batch in batches)
    {
        var tasks = batch.Select(async item =>
        {
            await semaphore.WaitAsync();
            try
            {
                var result = await ExtractTextAsync(item.Stream, item.Extension);
                results.Add(result);
            }
            finally
            {
                semaphore.Release();
            }
        });
        
        await Task.WhenAll(tasks);
    }
    
    return results.ToList();
}
```

```csharp
// Services/ChunkingService.cs
public class ChunkingService : IChunkingService
{
    private readonly ITokenizerService _tokenizer;
    private readonly ILogger<ChunkingService> _logger;
    
    public async Task<List<DocumentChunk>> ChunkTextAsync(
        string text, 
        Guid documentId,
        ChunkingOptions? options = null)
    {
        options ??= new ChunkingOptions();
        
        var chunks = options.Strategy switch
        {
            ChunkingStrategy.FixedSize => await ChunkFixedSizeAsync(text, options),
            ChunkingStrategy.RecursiveCharacter => await ChunkRecursiveAsync(text, options),
            ChunkingStrategy.Semantic => await ChunkSemanticAsync(text, options),
            ChunkingStrategy.DocumentStructure => await ChunkByStructureAsync(text, options),
            ChunkingStrategy.TokenBased => await ChunkByTokensAsync(text, options),
            _ => await ChunkRecursiveAsync(text, options)
        };
        
        // Assign to document
        foreach (var chunk in chunks)
        {
            chunk.DocumentId = documentId;
            chunk.ChunkHash = ComputeHash(chunk.Content);
            chunk.TokenCount = await _tokenizer.CountTokensAsync(chunk.Content);
            chunk.CharacterCount = chunk.Content.Length;
        }
        
        _logger.LogInformation(
            "Chunked document {DocId} into {ChunkCount} chunks", 
            documentId, chunks.Count);
        
        return chunks;
    }
    
    /// <summary>
    /// Recursive Character Splitting - RECOMMENDED for most use cases
    /// Splits text recursively using different separators, from largest to smallest
    /// </summary>
    private async Task<List<DocumentChunk>> ChunkRecursiveAsync(
        string text, 
        ChunkingOptions options)
    {
        var chunks = new List<DocumentChunk>();
        
        // Define separators in order of preference (largest unit first)
        var separators = new[] 
        { 
            "\n\n\n",   // Triple newline (section break)
            "\n\n",     // Double newline (paragraph)
            "\n",       // Single newline
            ". ",       // Sentence
            "; ",       // Clause
            ", ",       // Phrase
            " "         // Word (last resort)
        };
        
        var separatorLengths = separators.Select(s => s.Length).ToArray();
        
        // Recursive split function
        List<string> SplitText(string input, int separatorIndex = 0)
        {
            if (string.IsNullOrWhiteSpace(input))
                return new List<string>();
            
            var separator = separators[separatorIndex];
            var parts = input.Split(separator, StringSplitOptions.None)
                .Where(p => !string.IsNullOrWhiteSpace(p))
                .ToList();
            
            // Check if all parts are within size limit
            var allWithinLimit = true;
            var oversizedParts = new List<string>();
            
            foreach (var part in parts)
            {
                var tokenCount = await _tokenizer.CountTokensAsync(part);
                if (tokenCount > options.MaxChunkSize)
                {
                    allWithinLimit = false;
                    oversizedParts.Add(part);
                }
            }
            
            if (allWithinLimit)
            {
                return parts;
            }
            
            // Recursively split oversized parts with smaller separator
            var result = new List<string>();
            foreach (var part in parts)
            {
                var tokenCount = await _tokenizer.CountTokensAsync(part);
                if (tokenCount > options.MaxChunkSize && separatorIndex < separators.Length - 1)
                {
                    result.AddRange(SplitText(part, separatorIndex + 1));
                }
                else if (tokenCount >= options.MinChunkSize)
                {
                    result.Add(part);
                }
                else
                {
                    // Too small, merge with next
                    oversizedParts.Add(part);
                }
            }
            
            // Merge small parts
            return MergeSmallParts(result, options.MinChunkSize);
        }
        
        var rawChunks = SplitText(text);
        
        // Apply overlap
        chunks = ApplyOverlap(rawChunks, options.ChunkOverlap);
        
        return CreateChunks(chunks, options);
    }
    
    /// <summary>
    /// Semantic Chunking - Split by natural text boundaries
    /// Best for documents with clear paragraph structure
    /// </summary>
    private async Task<List<DocumentChunk>> ChunkSemanticAsync(
        string text, 
        ChunkingOptions options)
    {
        var chunks = new List<DocumentChunk>();
        
        // Split into paragraphs first
        var paragraphs = text.Split(new[] { "\n\n", "\r\n\r\n" }, StringSplitOptions.RemoveEmptyEntries)
            .Select(p => p.Trim())
            .Where(p => !string.IsNullOrWhiteSpace(p))
            .ToList();
        
        var currentChunk = new StringBuilder();
        var currentTokens = 0;
        var chunkIndex = 0;
        
        foreach (var paragraph in paragraphs)
        {
            var paragraphTokens = await _tokenizer.CountTokensAsync(paragraph);
            
            // If single paragraph exceeds max, do recursive split
            if (paragraphTokens > options.MaxChunkSize)
            {
                // Save current chunk first
                if (currentChunk.Length > 0)
                {
                    chunks.Add(CreateChunk(chunkIndex++, currentChunk.ToString(), text));
                    currentChunk.Clear();
                    currentTokens = 0;
                }
                
                // Recursively split the long paragraph
                var subChunks = await ChunkRecursiveAsync(paragraph, options);
                foreach (var subChunk in subChunks)
                {
                    chunks.Add(CreateChunk(chunkIndex++, subChunk.Content, text));
                }
            }
            // If adding this paragraph exceeds limit
            else if (currentTokens + paragraphTokens > options.MaxChunkSize)
            {
                if (currentChunk.Length > 0)
                {
                    chunks.Add(CreateChunk(chunkIndex++, currentChunk.ToString(), text));
                }
                
                // Start new chunk with overlap
                currentChunk.Clear();
                if (options.ChunkOverlap > 0 && chunks.Count > 0)
                {
                    var lastChunk = chunks.Last();
                    var lastParagraphs = lastChunk.Content.Split('\n');
                    if (lastParagraphs.Length > 0)
                    {
                        currentChunk.Append(lastParagraphs.Last() + "\n\n");
                    }
                }
                
                currentChunk.Append(paragraph);
                currentTokens = paragraphTokens;
            }
            else
            {
                if (currentChunk.Length > 0)
                {
                    currentChunk.Append("\n\n");
                }
                currentChunk.Append(paragraph);
                currentTokens += paragraphTokens;
            }
        }
        
        // Add remaining chunk
        if (currentChunk.Length > 0)
        {
            chunks.Add(CreateChunk(chunkIndex, currentChunk.ToString(), text));
        }
        
        return chunks;
    }
    
    /// <summary>
    /// Document Structure Chunking - Split by headings and sections
    /// Best for structured documents (manuals, reports)
    /// </summary>
    private async Task<List<DocumentChunk>> ChunkByStructureAsync(
        string text, 
        ChunkingOptions options)
    {
        var chunks = new List<DocumentChunk>();
        
        // Extract headings using regex
        var headingPattern = @"^(#{1,6})\s+(.+)$";
        var lines = text.Split('\n');
        
        var sections = new List<(string Heading, int Level, int StartLine, int EndLine)>();
        int currentLine = 0;
        string? currentHeading = null;
        int currentLevel = 0;
        int sectionStart = 0;
        
        foreach (var line in lines)
        {
            var match = Regex.Match(line.Trim(), headingPattern);
            if (match.Success)
            {
                // Save previous section
                if (currentHeading != null)
                {
                    sections.Add((currentHeading, currentLevel, sectionStart, currentLine - 1));
                }
                
                currentHeading = match.Groups[2].Value;
                currentLevel = match.Groups[1].Value.Length;
                sectionStart = currentLine;
            }
            currentLine++;
        }
        
        // Add last section
        if (currentHeading != null)
        {
            sections.Add((currentHeading, currentLevel, sectionStart, currentLine - 1));
        }
        
        // If no headings found, fall back to semantic chunking
        if (!sections.Any())
        {
            return await ChunkSemanticAsync(text, options);
        }
        
        // Create chunks from sections
        var allLines = lines.ToList();
        int chunkIndex = 0;
        var sectionPath = new StringBuilder();
        
        foreach (var section in sections)
        {
            var sectionLines = allLines.Skip(section.StartLine).Take(section.EndLine - section.StartLine + 1);
            var sectionText = string.Join("\n", sectionLines).Trim();
            
            // Update section path
            var parts = sectionPath.ToString().Split('.');
            while (parts.Length >= section.Level)
            {
                parts = parts.Take(parts.Length - 1).ToArray();
            }
            sectionPath.Clear();
            sectionPath.Append(string.Join(".", parts));
            if (sectionPath.Length > 0) sectionPath.Append('.');
            sectionPath.Append(section.Item1);
            
            // If section is too large, recursively chunk
            var sectionTokens = await _tokenizer.CountTokensAsync(sectionText);
            if (sectionTokens > options.MaxChunkSize)
            {
                var subChunks = await ChunkRecursiveAsync(sectionText, options);
                foreach (var subChunk in subChunks)
                {
                    chunks.Add(CreateChunk(chunkIndex++, subChunk.Content, text));
                }
            }
            else if (sectionTokens >= options.MinChunkSize)
            {
                chunks.Add(CreateChunk(chunkIndex++, sectionText, text));
            }
            else
            {
                // Merge with next section
                // (handled in post-processing)
            }
        }
        
        return chunks;
    }
    
    private List<DocumentChunk> ApplyOverlap(List<string> chunks, int overlapTokens)
    {
        if (overlapTokens <= 0 || chunks.Count <= 1)
            return chunks.Select((c, i) => CreateChunk(i, c, "")).ToList();
        
        var result = new List<DocumentChunk>();
        
        for (int i = 0; i < chunks.Count; i++)
        {
            var chunk = chunks[i];
            
            if (i > 0 && overlapTokens > 0)
            {
                // Add overlap from previous chunk
                var prevChunk = chunks[i - 1];
                var prevTokens = _tokenizer.EstimateTokens(prevChunk);
                
                if (prevTokens > overlapTokens)
                {
                    // Find where to start for overlap
                    var overlapText = GetOverlapText(
                        prevChunk, 
                        chunk, 
                        overlapTokens);
                    
                    chunk = overlapText + chunk;
                }
            }
            
            result.Add(CreateChunk(i, chunk, ""));
        }
        
        return result;
    }
    
    private DocumentChunk CreateChunk(int index, string content, string originalText)
    {
        return new DocumentChunk
        {
            Id = Guid.NewGuid(),
            ChunkIndex = index,
            Content = content,
            StartPosition = originalText.IndexOf(content, StringComparison.Ordinal),
            EndPosition = originalText.IndexOf(content, StringComparison.Ordinal) + content.Length,
            CreatedAt = DateTime.UtcNow
        };
    }
}
```

```csharp
public class ChunkingService : IChunkingService
{
    /// <summary>
    /// Extract and preserve tables as separate chunks
    /// Tables are often semantically important and should not be split
    /// </summary>
    public async Task<List<DocumentChunk>> ChunkWithTablePreservationAsync(
        string text, 
        ChunkingOptions options)
    {
        var chunks = new List<DocumentChunk>();
        var markdownText = text;
        
        // Extract tables (simple markdown format)
        var tablePattern = @"(\|.+\|\n)+";
        var tables = Regex.Matches(markdownText, tablePattern)
            .Cast<Match>()
            .Select(m => new 
            { 
                Table = m.Value, 
                Start = m.Index, 
                End = m.Index + m.Length 
            })
            .ToList();
        
        if (tables.Any())
        {
            int lastEnd = 0;
            int chunkIndex = 0;
            
            foreach (var table in tables)
            {
                // Chunk text before table
                var textBefore = markdownText.Substring(lastEnd, table.Start - lastEnd);
                if (!string.IsNullOrWhiteSpace(textBefore))
                {
                    var textChunks = await ChunkRecursiveAsync(textBefore, options);
                    foreach (var tc in textChunks)
                    {
                        chunks.Add(CreateChunk(chunkIndex++, tc.Content, markdownText));
                    }
                }
                
                // Add table as separate chunk
                var tableChunk = new DocumentChunk
                {
                    Id = Guid.NewGuid(),
                    ChunkIndex = chunkIndex++,
                    Content = $"[TABLE]\n{table.Table}\n[/TABLE]",
                    StartPosition = table.Start,
                    EndPosition = table.End,
                    CharacterCount = table.Table.Length,
                    TokenCount = await _tokenizer.CountTokensAsync(table.Table)
                };
                chunks.Add(tableChunk);
                
                lastEnd = table.End;
            }
            
            // Chunk remaining text
            if (lastEnd < markdownText.Length)
            {
                var remainingText = markdownText.Substring(lastEnd);
                var remainingChunks = await ChunkRecursiveAsync(remainingText, options);
                foreach (var rc in remainingChunks)
                {
                    chunks.Add(CreateChunk(chunkIndex++, rc.Content, markdownText));
                }
            }
        }
        else
        {
            // No tables, just chunk normally
            return await ChunkRecursiveAsync(text, options);
        }
        
        return chunks;
    }
    
    /// <summary>
    /// Detect and tag code blocks for special processing
    /// </summary>
    public List<DocumentChunk> TagCodeBlocks(List<DocumentChunk> chunks)
    {
        var codeBlockPattern = @"```[\s\S]*?```|`[^`]+`";
        
        foreach (var chunk in chunks)
        {
            if (Regex.IsMatch(chunk.Content, codeBlockPattern))
            {
                chunk.Keywords.Add("code");
                chunk.Quality = ChunkQuality.Good; // Code blocks are intentionally kept together
            }
        }
        
        return chunks;
    }
}
```

```csharp
private async Task<List<DocumentChunk>> SafeChunkAsync(
    string text, 
    Guid documentId,
    ChunkingOptions options)
{
    // 1. Empty text check
    if (string.IsNullOrWhiteSpace(text))
        throw new ChunkingException(ChunkingErrorCode.EmptyText, 
            "Cannot chunk empty text");
    
    // 2. Very short text - return as single chunk
    var totalTokens = await _tokenizer.CountTokensAsync(text);
    if (totalTokens < options.MinChunkSize)
    {
        return new List<DocumentChunk>
        {
            new DocumentChunk
            {
                DocumentId = documentId,
                Content = text,
                TokenCount = totalTokens,
                CharacterCount = text.Length,
                ChunkIndex = 0,
                Quality = ChunkQuality.Good
            }
        };
    }
    
    // 3. Handle encoding issues
    try
    {
        var normalizedText = NormalizeText(text);
        return await ChunkTextAsync(normalizedText, documentId, options);
    }
    catch (EncodingException ex)
    {
        _logger.LogWarning(ex, "Encoding issue in document {DocId}", documentId);
        // Fallback to simple character split
        return await ChunkFixedSizeAsync(text, options);
    }
}
```

**Các Edge Cases cần xử lý:**

2. **Text shorter than min chunk**: Return as single chunk
3. **All separators filtered**: Fall back to word-based split
4. **Unicode/encoding issues**: Normalize Unicode, handle BOM
5. **Very long words**: Split by character limit
6. **Mixed language content**: Preserve language boundaries
7. **Tables spanning multiple chunks**: Keep tables together
8. **Footnotes/endnotes**: Attach to relevant content

```csharp
// Services/EmbeddingService.cs
public class EmbeddingService : IEmbeddingService
{
    private readonly IOpenAiService _openAiService;
    private readonly ILocalEmbeddingService _localEmbeddingService;
    private readonly IRedisCache _cache;
    private readonly IEmbeddingRepository _embeddingRepository;
    private readonly IUnitOfWork _unitOfWork;
    private readonly ILogger<EmbeddingService> _logger;
    private readonly IOptions<EmbeddingSettings> _settings;
    
    public async Task<TextEmbedding> EmbedChunkAsync(DocumentChunk chunk)
    {
        // 1. Check cache first
        var cacheKey = $"embedding:{ComputeHash(chunk.Content)}";
        var cachedEmbedding = await _cache.GetAsync<TextEmbedding>(cacheKey);
        
        if (cachedEmbedding != null)
        {
            cachedEmbedding.IsFromCache = true;
            return cachedEmbedding;
        }
        
        // 2. Generate embedding
        TextEmbedding embedding;
        try
        {
            embedding = await GenerateEmbeddingAsync(chunk.Content);
            embedding.ChunkId = chunk.Id;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to generate embedding for chunk {ChunkId}", chunk.Id);
            throw new EmbeddingGenerationException("Failed to generate embedding", ex);
        }
        
        // 3. Cache the result
        await _cache.SetAsync(cacheKey, embedding, TimeSpan.FromDays(30));
        
        // 4. Save to database
        await _embeddingRepository.AddAsync(embedding);
        chunk.HasEmbedding = true;
        chunk.EmbeddingId = embedding.Id;
        
        await _unitOfWork.SaveChangesAsync();
        
        return embedding;
    }
    
    public async Task<EmbeddingBatchResult> EmbedBatchAsync(EmbeddingBatchRequest request)
    {
        var result = new EmbeddingBatchResult
        {
            Embeddings = new List<EmbeddingResult>()
        };
        
        var stopwatch = Stopwatch.StartNew();
        
        // 1. Deduplicate texts
        var uniqueTexts = request.Texts
            .Select((text, index) => new { text, index, hash = ComputeHash(text) })
            .GroupBy(x => x.hash)
            .Select(g => g.First())
            .ToList();
        
        // 2. Check cache for each unique text
        var uncachedTexts = new List<(int OriginalIndex, string Text, string Hash)>();
        var cachedResults = new Dictionary<int, EmbeddingResult>();
        
        foreach (var item in uniqueTexts)
        {
            var cacheKey = $"embedding:{item.hash}";
            var cached = await _cache.GetAsync<EmbeddingResult>(cacheKey);
            
            if (cached != null)
            {
                cachedResults[item.index] = cached;
            }
            else
            {
                uncachedTexts.Add((item.index, item.text, item.hash));
            }
        }
        
        // 3. Batch API calls (with retry)
        var batchSize = _settings.Value.BatchSize; // 100 for OpenAI
        for (int i = 0; i < uncachedTexts.Count; i += batchSize)
        {
            var batch = uncachedTexts.Skip(i).Take(batchSize).ToList();
            
            try
            {
                var batchResult = await CallEmbeddingApiWithRetryAsync(
                    batch.Select(b => b.Text).ToList(),
                    request.Model);
                
                for (int j = 0; j < batch.Count; j++)
                {
                    var embeddingResult = batchResult[j];
                    embeddingResult.Index = batch[j].OriginalIndex;
                    
                    result.Embeddings.Add(embeddingResult);
                    
                    // Cache the result
                    var cacheKey = $"embedding:{batch[j].Hash}";
                    await _cache.SetAsync(cacheKey, embeddingResult, TimeSpan.FromDays(30));
                }
                
                result.TotalTokens += batchResult.Sum(e => e.Vector.Length);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Batch embedding failed for indices {Indices}", 
                    string.Join(",", batch.Select(b => b.OriginalIndex)));
                
                // Add error results
                foreach (var item in batch)
                {
                    result.Errors.Add(new EmbeddingError
                    {
                        Index = item.OriginalIndex,
                        Text = item.Text,
                        ErrorCode = "API_ERROR",
                        Message = ex.Message
                    });
                }
            }
        }
        
        // 4. Add cached results
        foreach (var cached in cachedResults)
        {
            result.Embeddings.Insert(cached.Key, cached.Value);
        }
        
        // 5. Calculate cost
        result.TotalCostUsd = CalculateCost(result.TotalTokens, request.Model);
        result.ProcessingTime = stopwatch.Elapsed;
        
        return result;
    }
    
    private async Task<List<EmbeddingResult>> CallEmbeddingApiWithRetryAsync(
        List<string> texts,
        EmbeddingModel model)
    {
        var maxRetries = 3;
        var delay = TimeSpan.FromSeconds(1);
        
        for (int attempt = 1; attempt <= maxRetries; attempt++)
        {
            try
            {
                var openAiModel = model switch
                {
                    EmbeddingModel.OpenAI_text_embedding_3_small => "text-embedding-3-small",
                    EmbeddingModel.OpenAI_text_embedding_3_large => "text-embedding-3-large",
                    EmbeddingModel.OpenAI_text_embedding_ada_002 => "text-embedding-ada-002",
                    _ => "text-embedding-3-small"
                };
                
                return await _openAiService.CreateEmbeddingsAsync(openAiModel, texts);
            }
            catch (OpenAiRateLimitException ex)
            {
                if (attempt == maxRetries)
                    throw;
                
                // Exponential backoff
                var waitTime = delay * Math.Pow(2, attempt - 1);
                _logger.LogWarning(
                    "Rate limited. Retrying in {Seconds}s (attempt {Attempt}/{MaxRetries})",
                    waitTime.TotalSeconds, attempt, maxRetries);
                
                await Task.Delay(waitTime);
            }
            catch (OpenAiServiceUnavailableException ex)
            {
                if (attempt == maxRetries)
                    throw;
                
                // Server error - wait and retry
                await Task.Delay(delay * attempt);
            }
        }
        
        throw new Exception("Should not reach here");
    }
    
    private double CalculateCost(int tokenCount, EmbeddingModel model)
    {
        // OpenAI pricing (as of 2024)
        return model switch
        {
            EmbeddingModel.OpenAI_text_embedding_3_small => tokenCount * 0.00002 / 1000, // $0.02/1M tokens
            EmbeddingModel.OpenAI_text_embedding_3_large => tokenCount * 0.00013 / 1000, // $0.13/1M tokens
            EmbeddingModel.OpenAI_text_embedding_ada_002 => tokenCount * 0.0001 / 1000, // $0.10/1M tokens
            _ => 0 // Local models are free
        };
    }
}
```

```csharp
public async Task<List<TextEmbedding>> SafeEmbedBatchAsync(
    List<DocumentChunk> chunks)
{
    var results = new List<TextEmbedding>();
    var errors = new List<EmbeddingError>();
    
    // 1. Check for empty chunks
    chunks = chunks.Where(c => !string.IsNullOrWhiteSpace(c.Content)).ToList();
    
    if (!chunks.Any())
        return results;
    
    // 2. Check total batch size
    if (chunks.Count > 1000)
    {
        // Split into smaller batches
        var batches = chunks.Chunk(100);
        foreach (var batch in batches)
        {
            var batchResults = await SafeEmbedBatchAsync(batch);
            results.AddRange(batchResults);
        }
        return results;
    }
    
    // 3. Try primary service
    try
    {
        var request = new EmbeddingBatchRequest
        {
            Texts = chunks.Select(c => c.Content).ToList()
        };
        
        var result = await EmbedBatchAsync(request);
        
        if (result.Errors.Any())
        {
            errors.AddRange(result.Errors);
            _logger.LogWarning(
                "Batch embedding had {ErrorCount} errors out of {TotalCount}",
                result.Errors.Count, chunks.Count);
        }
        
        // Map results to embeddings
        for (int i = 0; i < result.Embeddings.Count; i++)
        {
            results.Add(new TextEmbedding
            {
                Id = Guid.NewGuid(),
                ChunkId = chunks[i].Id,
                Vector = result.Embeddings[i].Vector,
                Dimensions = result.Embeddings[i].Vector.Length,
                Model = _settings.Value.PrimaryModel,
                GeneratedAt = DateTime.UtcNow
            });
        }
    }
    catch (OpenAiApiException ex) when (ex.IsRateLimit)
    {
        _logger.LogWarning("Rate limited. Falling back to local embedding");
        return await EmbedWithLocalModelAsync(chunks);
    }
    catch (OpenAiApiException ex) when (ex.IsServerError)
    {
        _logger.LogError(ex, "OpenAI server error. Retrying with local model");
        return await EmbedWithLocalModelAsync(chunks);
    }
    
    return results;
}
```

#### 6.3.1 Hybrid Search

```csharp
// Services/SearchService.cs
public class SearchService : ISearchService
{
    private readonly ISemanticSearchProvider _semanticSearch;
    private readonly IKeywordSearchProvider _keywordSearch;
    private readonly IRerankerService _reranker;
    private readonly IEmbeddingService _embeddingService;
    private readonly ICacheService _cache;
    private readonly ILogger<SearchService> _logger;
    
    public async Task<SearchResponse> SearchAsync(SearchQuery query)
    {
        var stopwatch = Stopwatch.StartNew();
        
        // 1. Generate query embedding
        var queryEmbedding = await _embeddingService.EmbedTextAsync(query.Query);
        
        // 2. Execute searches based on type
        List<SearchResult> semanticResults = new();
        List<SearchResult> keywordResults = new();
        
        if (query.SearchType == SearchType.Semantic || query.SearchType == SearchType.Hybrid)
        {
            semanticResults = await _semanticSearch.SearchAsync(
                queryEmbedding,
                query.TopK * 2, // Get more for reranking
                query);
        }
        
        if (query.SearchType == SearchType.Keyword || query.SearchType == SearchType.Hybrid)
        {
            keywordResults = await _keywordSearch.SearchAsync(
                query.Query,
                query.TopK * 2,
                query);
        }
        
        // 3. Combine results
        List<SearchResult> combinedResults;
        if (query.SearchType == SearchType.Hybrid)
        {
            combinedResults = CombineResults(semanticResults, keywordResults);
        }
        else if (query.SearchType == SearchType.Semantic)
        {
            combinedResults = semanticResults;
        }
        else
        {
            combinedResults = keywordResults;
        }
        
        // 4. Apply filters
        combinedResults = ApplyFilters(combinedResults, query);
        
        // 5. Rerank (if enabled)
        if (combinedResults.Count > query.TopK && query.SearchType == SearchType.Hybrid)
        {
            combinedResults = await _reranker.RerankAsync(
                query.Query,
                combinedResults.Take(query.TopK * 3).ToList());
        }
        
        // 6. Paginate
        var paginatedResults = combinedResults
            .Skip(query.Skip)
            .Take(query.TopK)
            .ToList();
        
        // 7. Generate highlights
        if (query.IncludeHighlights)
        {
            foreach (var result in paginatedResults)
            {
                result.HighlightedContent = GenerateHighlight(result.ChunkContent, query.Query);
                result.MatchedTerms = ExtractMatchedTerms(result.ChunkContent, query.Query);
            }
        }
        
        stopwatch.Stop();
        
        return new SearchResponse
        {
            Results = paginatedResults,
            TotalCount = combinedResults.Count,
            QueryTimeMs = stopwatch.ElapsedMilliseconds,
            QueryEmbeddingModel = "text-embedding-3-small",
            SearchType = query.SearchType.ToString(),
            Pagination = new PaginationInfo
            {
                Skip = query.Skip,
                TopK = query.TopK,
                TotalCount = combinedResults.Count,
                NextCursor = combinedResults.Count > query.Skip + query.TopK
                    ? GenerateCursor(query.Skip + query.TopK)
                    : null
            }
        };
    }
    
    private List<SearchResult> CombineResults(
        List<SearchResult> semantic, 
        List<SearchResult> keyword)
    {
        var combined = new Dictionary<Guid, SearchResult>();
        
        // Add semantic results with score
        foreach (var result in semantic)
        {
            result.SemanticScore = result.CombinedScore;
            combined[result.ChunkId] = result;
        }
        
        // Add/enhance with keyword results
        foreach (var result in keyword)
        {
            if (combined.ContainsKey(result.ChunkId))
            {
                combined[result.ChunkId].KeywordScore = result.CombinedScore;
            }
            else
            {
                result.KeywordScore = result.CombinedScore;
                combined[result.ChunkId] = result;
            }
        }
        
        // Calculate combined score (RRF - Reciprocal Rank Fusion)
        // This is a common technique for combining rankings
        const int k = 60; // RRF parameter
        foreach (var result in combined.Values)
        {
            var semanticRank = semantic.FindIndex(r => r.ChunkId == result.ChunkId) + 1;
            var keywordRank = keyword.FindIndex(r => r.ChunkId == result.ChunkId) + 1;
            
            result.CombinedScore = 
                (semanticRank > 0 ? 1.0 / (k + semanticRank) : 0) * 0.6 + // 60% weight to semantic
                (keywordRank > 0 ? 1.0 / (k + keywordRank) : 0) * 0.4;    // 40% weight to keyword
        }
        
        return combined.Values
            .OrderByDescending(r => r.CombinedScore)
            .ToList();
    }
    
    private List<SearchResult> ApplyFilters(List<SearchResult> results, SearchQuery query)
    {
        if (query.CategoryIds?.Any() == true)
        {
            results = results.Where(r => 
                query.CategoryIds.Contains(r.DocumentId)).ToList();
        }
        
        if (query.DateFrom.HasValue)
        {
            results = results.Where(r => 
                r.DocumentDate >= query.DateFrom).ToList();
        }
        
        if (query.DateTo.HasValue)
        {
            results = results.Where(r => 
                r.DocumentDate <= query.DateTo).ToList();
        }
        
        if (query.UploaderId.HasValue)
        {
            // Filter by uploader - need to join with document
            // This would be handled in the query phase for better performance
        }
        
        return results;
    }
    
    private string GenerateHighlight(string content, string query)
    {
        // Find query terms in content and wrap with <mark> tags
        var terms = query.Split(' ', StringSplitOptions.RemoveEmptyEntries);
        var highlighted = content;
        
        foreach (var term in terms.Where(t => t.Length > 2))
        {
            var pattern = $@"({Regex.Escape(term)})";
            highlighted = Regex.Replace(
                highlighted, 
                pattern, 
                "<mark>$1</mark>", 
                RegexOptions.IgnoreCase);
        }
        
        // Limit snippet length
        if (highlighted.Length > 500)
        {
            var firstMark = highlighted.IndexOf("<mark>");
            if (firstMark > 200)
            {
                highlighted = "..." + highlighted.Substring(firstMark - 100);
            }
            highlighted = highlighted.Substring(0, 500) + "...";
        }
        
        return highlighted;
    }
```

```csharp
// Infrastructure/Search/Bm25SearchProvider.cs
public class Bm25SearchProvider : IKeywordSearchProvider
{
    private readonly SearchDbContext _context;
    
    public async Task<List<SearchResult>> SearchAsync(
        string query,
        int topK,
        SearchQuery searchQuery)
    {
        // Tokenize query
        var terms = Tokenize(query);
        
        // Build BM25 query using tsvector
        var tsQuery = string.Join(" & ", terms.Select(t => $"{t}:*"));
        
        var results = await _context.DocumentChunks
            .Join(_context.Documents, c => c.DocumentId, d => d.Id, (c, d) => new { Chunk = c, Doc = d })
            .Where(x => x.Chunk.SearchVector.Matches(EF.Functions.PlainToTsQuery("english", tsQuery)))
            .Where(x => x.Doc.IsPublic && x.Doc.Status == DocumentStatus.Processed)
            .OrderByDescending(x => x.Chunk.SearchVector.Rank(
                EF.Functions.PlainToTsQuery("english", tsQuery)))
            .Take(topK)
            .Select(x => new SearchResult
            {
                ChunkId = x.Chunk.Id,
                DocumentId = x.Doc.Id,
                ChunkContent = x.Chunk.Content,
                DocumentTitle = x.Doc.Title,
                KeywordScore = x.Chunk.SearchVector.Rank(
                    EF.Functions.PlainToTsQuery("english", tsQuery)),
                CombinedScore = x.Chunk.SearchVector.Rank(
                    EF.Functions.PlainToTsQuery("english", tsQuery))
            })
            .ToListAsync();
        
        return results;
    }
}
```

```csharp
// Handle edge cases
public async Task<SearchResponse> SafeSearchAsync(SearchQuery query)
{
    // 1. Empty query
    if (string.IsNullOrWhiteSpace(query.Query))
        throw new SearchException(SearchErrorCode.EmptyQuery, 
            "Từ khóa tìm kiếm không được để trống");
    
    // 2. Query too short
    if (query.Query.Length < 2)
        throw new SearchException(SearchErrorCode.QueryTooShort,
            "Từ khóa quá ngắn. Vui lòng nhập ít nhất 2 ký tự");
    
    // 3. Query too long (truncate)
    if (query.Query.Length > 1000)
        query.Query = query.Query.Substring(0, 1000);
    
    // 4. Invalid date range
    if (query.DateFrom.HasValue && query.DateTo.HasValue && 
        query.DateFrom > query.DateTo)
    {
        throw new SearchException(SearchErrorCode.InvalidFilters,
            "Ngày bắt đầu phải nhỏ hơn ngày kết thúc");
    }
    
    // 5. Timeout handling
    using var cts = new CancellationTokenSource(TimeSpan.FromSeconds(10));
    
    try
    {
        return await SearchAsync(query);
    }
    catch (OperationCanceledException)
    {
        _logger.LogWarning("Search timed out for query: {Query}", query.Query);
        throw new SearchException(SearchErrorCode.SearchTimeout,
            "Tìm kiếm quá thời gian. Vui lòng thử lại");
    }
}
```

**Các Edge Cases cần xử lý:**

1. **Empty index**: Return "No documents indexed yet" message
2. **Newly indexed documents**: Allow 1-2 minute delay for indexing
3. **Deleted documents**: Remove from search index immediately
4. **Very large result sets**: Cap at 1000, use cursor pagination
5. **Special characters in query**: Escape properly
6. **Multiple languages**: Handle mixed-language queries
7. **Synonym expansion**: Expand common terms (e.g., "HĐLĐ" → "hợp đồng lao động")

```csharp
    private readonly ISearchService _searchService;
    private readonly ILlmService _llmService;
    private readonly ISessionRepository _sessionRepository;
    private readonly IMessageRepository _messageRepository;
    private readonly IUnitOfWork _unitOfWork;
    private readonly ILogger<ChatService> _logger;
    private readonly IOptions<ChatSettings> _settings;
    
    public async Task<ChatResponse> ChatAsync(ChatRequest request, Guid userId)
    {
        var stopwatch = Stopwatch.StartNew();
        
        // 1. Get or create session
        var session = await GetOrCreateSessionAsync(request.SessionId, userId);
        var messageIndex = session.MessageCount;
        
        // 2. Create user message
        var userMessage = new ChatMessage
        {
            Id = Guid.NewGuid(),
            SessionId = session.Id,
            MessageIndex = messageIndex,
            Role = MessageRole.User,
            Content = request.Question,
            Status = MessageStatus.Completed
        };
        
        await _messageRepository.AddAsync(userMessage);
        
        // 3. Retrieve relevant context
        var retrievalStart = DateTime.UtcNow;
        var retrievedChunks = await RetrieveContextAsync(
            request.Question,
            request.TopK,
            request.SimilarityThreshold,
            request.DocumentIds);
        
        var retrievalTime = (DateTime.UtcNow - retrievalStart).TotalMilliseconds;
        
        // 4. Build prompt with context
        var prompt = BuildPrompt(request.Question, retrievedChunks);
        
        // 5. Call LLM
        var llmResponse = await CallLlmAsync(prompt, session.Model);
        
        // 6. Generate citations
        var citations = GenerateCitations(retrievedChunks);
        
        // 7. Generate follow-up questions
        var followUps = request.IncludeFollowUp 
            ? await GenerateFollowUpQuestionsAsync(request.Question, retrievedChunks)
            : new List<string>();
        
        // 8. Create assistant message
        var assistantMessage = new ChatMessage
        {
            Id = Guid.NewGuid(),
            SessionId = session.Id,
            MessageIndex = messageIndex + 1,
            Role = MessageRole.Assistant,
            Content = llmResponse.Content,
            Citations = citations,
            TokensUsed = llmResponse.TotalTokens,
            ModelUsed = llmResponse.Model,
            LatencyMs = stopwatch.ElapsedMilliseconds,
            Status = MessageStatus.Completed
        };
        
        await _messageRepository.AddAsync(assistantMessage);
        
        // 9. Update session
        session.LastMessageAt = DateTime.UtcNow;
        session.MessageCount += 2;
        session.TokensUsed += llmResponse.TotalTokens;
        
        // Update session title from first question
        if (messageIndex == 0)
        {
            session.Title = Truncate(request.Question, 100);
        }
        
        await _unitOfWork.SaveChangesAsync();
        
        stopwatch.Stop();
        
        return new ChatResponse
        {
            SessionId = session.Id,
            MessageId = assistantMessage.Id,
            Answer = llmResponse.Content,
            Citations = citations,
            FollowUpQuestions = followUps,
            Metadata = new ChatMetadata
            {
                ModelUsed = llmResponse.Model,
                TokensUsed = llmResponse.TotalTokens,
                PromptTokens = llmResponse.PromptTokens,
                CompletionTokens = llmResponse.CompletionTokens,
                LatencyMs = stopwatch.ElapsedMilliseconds,
                ChunksRetrieved = retrievedChunks.Count,
                AverageSimilarity = retrievedChunks.Average(c => c.SimilarityScore)
            }
        };
    }
    
    private async Task<List<RetrievedChunk>> RetrieveContextAsync(
        string question,
        int topK,
        double similarityThreshold,
        List<Guid>? documentIds)
    {
        // 1. Semantic search
        var searchQuery = new SearchQuery
        {
            Query = question,
            TopK = topK * 2, // Get more for filtering
            DocumentIds = documentIds,
            IncludeHighlights = true
        };
        
        var searchResults = await _searchService.SearchAsync(searchQuery);
        
        // 2. Filter by similarity threshold
        var relevantChunks = searchResults.Results
            .Where(r => r.SemanticScore >= similarityThreshold)
            .Take(topK)
            .Select(r => new RetrievedChunk
            {
                ChunkId = r.ChunkId,
                DocumentId = r.DocumentId,
                DocumentTitle = r.DocumentTitle,
                Content = r.ChunkContent,
                PageNumber = r.PageNumber,
                SimilarityScore = r.SemanticScore,
                SectionPath = r.SectionPath
            })
            .ToList();
        
        // 3. Deduplicate by document (keep best chunk per document)
        var deduplicated = relevantChunks
            .GroupBy(c => c.DocumentId)
            .Select(g => g.OrderByDescending(c => c.SimilarityScore).First())
            .OrderByDescending(c => c.SimilarityScore)
            .Take(topK)
            .ToList();
        
        return deduplicated;
    }
    
    private string BuildPrompt(string question, List<RetrievedChunk> chunks)
    {
        var systemPrompt = @"Bạn là một trợ lý AI chuyên trả lời câu hỏi dựa trên tài liệu được cung cấp.

**QUY TẮC QUAN TRỌNG:**
1. CHỈ sử dụng thông tin từ tài liệu được cung cấp trong phần [CONTEXT]
2. Nếu câu hỏi không thể trả lời từ context, hãy nói rõ: ""Tôi không tìm thấy thông tin này trong tài liệu được cung cấp.""
3. TRÍCH DẪN nguồn cho mỗi thông tin bằng định dạng [Tên tài liệu, Trang X]
4. TRẢ LỜI ngắn gọn, rõ ràng, sử dụng tiếng Việt
5. KHÔNG suy đoán hay bịa đặt thông tin không có trong tài liệu
6. Nếu có nhiều nguồn, ưu tiên nguồn có độ tương đồng cao hơn

**Định dạng trả lời:**
- Trả lời: [Nội dung câu trả lời]
- Nguồn: [Danh sách tài liệu tham khảo]";

        var contextSection = new StringBuilder();
        contextSection.AppendLine("[CONTEXT]");
        
        for (int i = 0; i < chunks.Count; i++)
        {
            var chunk = chunks[i];
            var reference = $"[{chunk.DocumentTitle}" +
                (chunk.PageNumber.HasValue ? $", Trang {chunk.PageNumber}" : "") + 
                $"]";
            
            contextSection.AppendLine($"--- Document {i + 1} {reference} ---");
            contextSection.AppendLine(chunk.Content);
            contextSection.AppendLine();
        }
        
        contextSection.AppendLine("[/CONTEXT]");
        contextSection.AppendLine();
        contextSection.AppendLine($"Câu hỏi: {question}");
        
        return systemPrompt + "\n\n" + contextSection.ToString();
    }
    
    private List<MessageCitation> GenerateCitations(List<RetrievedChunk> chunks)
    {
        return chunks.Select(c => new MessageCitation
        {
            ChunkId = c.ChunkId,
            DocumentId = c.DocumentId,
            DocumentTitle = c.DocumentTitle,
            PageNumber = c.PageNumber,
            Content = c.Content.Substring(0, Math.Min(200, c.Content.Length)) + "...",
            SimilarityScore = c.SimilarityScore,
            Reference = $"[{c.DocumentTitle}" +
                (c.PageNumber.HasValue ? $", Trang {c.PageNumber}" : "") + "]"
        }).ToList();
    }
}
```

```csharp
// Services/LlmService.cs
public class LlmService : ILlmService
{
    private readonly IOpenAiService _openAiService;
    private readonly IClaudeService _claudeService;
    private readonly IUsageTracker _usageTracker;
    private readonly ILogger<LlmService> _logger;
    
    public async Task<LlmResponse> CallLlmAsync(string prompt, ChatModel model)
    {
        var stopwatch = Stopwatch.StartNew();
        
        LlmResponse response;
        
        switch (model)
        {
            case ChatModel.GPT4o:
            case ChatModel.GPT4o_mini:
            case ChatModel.GPT4_Turbo:
                response = await CallOpenAiAsync(prompt, model);
                break;
                
            case ChatModel.Claude3_Sonnet:
            case ChatModel.Claude3_Haiku:
                response = await CallClaudeAsync(prompt, model);
                break;
                
            default:
                response = await CallOpenAiAsync(prompt, ChatModel.GPT4o);
                break;
        }
        
        stopwatch.Stop();
        response.LatencyMs = stopwatch.ElapsedMilliseconds;
        
        // Track usage
        await _usageTracker.RecordUsageAsync(new UsageRecord
        {
            Model = model.ToString(),
            PromptTokens = response.PromptTokens,
            CompletionTokens = response.CompletionTokens,
            LatencyMs = response.LatencyMs.Value
        });
        
        return response;
    }
    
    private async Task<LlmResponse> CallOpenAiAsync(string prompt, ChatModel model)
    {
        var modelName = model switch
        {
            ChatModel.GPT4o => "gpt-4o",
            ChatModel.GPT4o_mini => "gpt-4o-mini",
            ChatModel.GPT4_Turbo => "gpt-4-turbo",
            _ => "gpt-4o"
        };
        
        var request = new ChatCompletionRequest
        {
            Model = modelName,
            Messages = new List<ChatMessage>
            {
                new ChatMessage { Role = "user", Content = prompt }
            },
            Temperature = 0.7,
            MaxTokens = 4000,
            ResponseFormat = new ResponseFormat { Type = "text" }
        };
        
        try
        {
            var response = await _openAiService.CreateChatCompletionAsync(request);
            
            return new LlmResponse
            {
                Content = response.Choices.First().Message.Content,
                Model = response.Model,
                PromptTokens = response.Usage.PromptTokens,
                CompletionTokens = response.Usage.CompletionTokens,
                TotalTokens = response.Usage.TotalTokens
            };
        }
        catch (OpenAiRateLimitException ex)
        {
            _logger.LogWarning("Rate limited, retrying with backoff");
            await Task.Delay(TimeSpan.FromSeconds(5));
            return await CallOpenAiAsync(prompt, model);
        }
    }
}
```

```csharp
public class ConversationHistoryManager
{
    private readonly IMessageRepository _messageRepo;
    private readonly ICacheService _cache;
    private readonly int MaxContextMessages = 20; // Keep last 20 messages
    
    public async Task<List<ChatMessage>> GetContextMessagesAsync(
        Guid sessionId,
        int maxTokens = 8000)
    {
        // Try cache first
        var cacheKey = $"chat:context:{sessionId}";
        var cached = await _cache.GetAsync<List<ChatMessage>>(cacheKey);
        if (cached != null)
            return cached;
        
        // Get from database
        var messages = await _messageRepo.GetRecentMessagesAsync(sessionId, MaxContextMessages);
        
        // Truncate if too long
        var truncated = await TruncateToTokenLimitAsync(messages, maxTokens);
        
        // Cache for 5 minutes
        await _cache.SetAsync(cacheKey, truncated, TimeSpan.FromMinutes(5));
        
        return truncated;
    }
    
    private async Task<List<ChatMessage>> TruncateToTokenLimitAsync(
        List<ChatMessage> messages,
        int maxTokens)
    {
        var result = new List<ChatMessage>();
        var totalTokens = 0;
        
        // Add system message first
        var systemMessage = messages.FirstOrDefault(m => m.Role == MessageRole.System);
        if (systemMessage != null)
        {
            result.Add(systemMessage);
        }
        
        // Add messages from newest to oldest
        foreach (var message in messages.Where(m => m.Role != MessageRole.System).Reverse())
        {
            var messageTokens = await _tokenizer.CountTokensAsync(message.Content);
            if (totalTokens + messageTokens > maxTokens)
                break;
            
            result.Insert(systemMessage != null ? 1 : 0, message);
            totalTokens += messageTokens;
        }
        
        return result;
    }
}
```

```csharp
// Services/SuggestionService.cs
public class SuggestionService : ISuggestionService
{
    private readonly ISearchService _searchService;
    private readonly IRedisCache _cache;
    private readonly IQueryLogRepository _queryLogRepo;
    private readonly ILlmService _llmService;
    private readonly ILogger<SuggestionService> _logger;
    
    public async Task<List<SearchSuggestion>> GetSuggestionsAsync(
        string query,
        int maxResults = 10,
        string? userId = null)
    {
        var suggestions = new List<SearchSuggestion>();
        
        if (string.IsNullOrWhiteSpace(query))
        {
            // Return popular queries when no query
            return await GetPopularQueriesAsync(maxResults);
        }
        
        // 1. Autocomplete (prefix match)
        var autocomplete = await GetAutocompleteSuggestionsAsync(query, maxResults / 2);
        suggestions.AddRange(autocomplete);
        
        // 2. Spelling correction
        var corrected = await GetCorrectedQueryAsync(query);
        if (corrected != null && corrected.Text != query)
        {
            suggestions.Add(corrected);
        }
        
        // 3. Related queries (from LLM)
        if (query.Length > 3)
        {
            var related = await GetRelatedQueriesAsync(query, maxResults);
            suggestions.AddRange(related);
        }
        
        // 4. Query expansion (optional)
        var expanded = await GetExpandedQueriesAsync(query, maxResults / 2);
        suggestions.AddRange(expanded);
        
        // Log query for analytics
        await LogQueryAsync(query, userId);
        
        // Remove duplicates and sort
        return suggestions
            .GroupBy(s => s.Text.ToLowerInvariant())
            .Select(g => g.First())
            .OrderByDescending(s => s.Score)
            .Take(maxResults)
            .ToList();
    }
    
    private async Task<List<SearchSuggestion>> GetAutocompleteSuggestionsAsync(
        string query,
        int maxResults)
    {
        // Check cache first
        var cacheKey = $"suggest:autocomplete:{query.ToLowerInvariant()}";
        var cached = await _cache.GetAsync<List<SearchSuggestion>>(cacheKey);
        if (cached != null)
            return cached.Take(maxResults).ToList();
        
        // Query historical searches for prefix match
        var suggestions = await _queryLogRepo.GetByPrefixAsync(query, maxResults * 2);
        
        var results = suggestions
            .GroupBy(q => q.Query.ToLowerInvariant())
            .Select(g => new SearchSuggestion
            {
                Text = g.First().Query,
                Score = g.Count(), // Popularity score
                ResultCount = g.Sum(q => q.ResultCount),
                Type = SuggestionType.Autocomplete,
                Highlights = HighlightMatch(g.First().Query, query)
            })
            .OrderByDescending(s => s.Score)
            .Take(maxResults)
            .ToList();
        
        // Cache for 5 minutes
        await _cache.SetAsync(cacheKey, results, TimeSpan.FromMinutes(5));
        
        return results;
    }
    
    private async Task<SearchSuggestion?> GetCorrectedQueryAsync(string query)
    {
        // Simple spelling correction using edit distance
        var words = query.Split(' ', StringSplitOptions.RemoveEmptyEntries);
        var corrections = new List<string>();
        
        foreach (var word in words)
        {
            if (word.Length < 3)
            {
                corrections.Add(word);
                continue;
            }
            
            // Check if word exists in dictionary
            var exists = await _queryLogRepo.WordExistsAsync(word);
            if (!exists)
            {
                // Find similar words
                var similar = await _queryLogRepo.FindSimilarWordsAsync(word, 1);
                if (similar.Any())
                {
                    corrections.Add(similar.First());
                }
                else
                {
                    corrections.Add(word);
                }
            }
            else
            {
                corrections.Add(word);
            }
        }
        
        var correctedQuery = string.Join(" ", corrections);
        
        if (correctedQuery != query)
        {
            // Verify corrected query has results
            var searchResult = await _searchService.SearchAsync(new SearchQuery
            {
                Query = correctedQuery,
                TopK = 1
            });
            
            if (searchResult.TotalCount > 0)
            {
                return new SearchSuggestion
                {
                    Text = correctedQuery,
                    Score = 0.8,
                    ResultCount = searchResult.TotalCount,
                    Type = SuggestionType.CorrectedQuery,
                    Highlights = corrections
                };
            }
        }
        
        return null;
    }
    
    private async Task<List<SearchSuggestion>> GetRelatedQueriesAsync(
        string query,
        int maxResults)
    {
        // Use LLM to generate related queries
        var prompt = $@"Dựa trên truy vấn tìm kiếm: ""{query}""
Hãy đề xuất 5 truy vấn tìm kiếm liên quan mà người dùng có thể quan tâm.
Mỗi truy vấn nên ngắn gọn (dưới 10 từ) và liên quan đến chủ đề.
Trả lời theo định dạng JSON array:
[""query 1"", ""query 2"", ""query 3"", ""query 4"", ""query 5""]";

        try
        {
            var response = await _llmService.CallLlmAsync(prompt, ChatModel.GPT4o_mini);
            
            // Parse JSON response
            var relatedQueries = JsonSerializer.Deserialize<List<string>>(
                response.Content.Trim(),
                new JsonSerializerOptions { PropertyNameCaseInsensitive = true });
            
            if (relatedQueries == null)
                return new List<SearchSuggestion>();
            
            var suggestions = new List<SearchSuggestion>();
            foreach (var relatedQuery in relatedQueries.Take(maxResults))
            {
                var searchResult = await _searchService.SearchAsync(new SearchQuery
                {
                    Query = relatedQuery,
                    TopK = 1
                });
                
                suggestions.Add(new SearchSuggestion
                {
                    Text = relatedQuery,
                    Score = 0.7,
                    ResultCount = searchResult.TotalCount,
                    Type = SuggestionType.RelatedQuery
                });
            }
            
            return suggestions;
        }
        catch (Exception ex)
        {
            _logger.LogWarning(ex, "Failed to get related queries for: {Query}", query);
            return new List<SearchSuggestion>();
        }
    }
    
    private List<string> HighlightMatch(string suggestion, string query)
    {
        var highlights = new List<string>();
        var suggestionLower = suggestion.ToLowerInvariant();
        var queryLower = query.ToLowerInvariant();
        
        if (suggestionLower.StartsWith(queryLower))
        {
            highlights.Add(suggestion.Substring(0, query.Length));
        }
        
        return highlights;
    }
}
```

#### Chart Data Generation

```csharp
public async Task<UsageChartData> GetUsageChartAsync(DateRange range)
{
    var data = new List<UsageDataPoint>();
    
    for (var date = range.StartDate; date <= range.EndDate; date = date.AddDays(1))
    {
        var dayData = await _usageRepo.GetDailyStatsAsync(date);
        
        data.Add(new UsageDataPoint
        {
            Date = date,
            Searches = dayData.SearchCount,
            Chats = dayData.ChatCount,
            Uploads = dayData.DocumentUploads,
            ActiveUsers = dayData.ActiveUsers,
            AiCost = dayData.AiCostUsd
        });
    }
    
    return new UsageChartData
    {
        DataPoints = data,
        TotalSearches = data.Sum(d => d.Searches),
        TotalChats = data.Sum(d => d.Chats),
        TotalUploads = data.Sum(d => d.Uploads),
        TotalCost = data.Sum(d => d.AiCost),
        AverageDailyUsers = data.Average(d => d.ActiveUsers)
    };
}
```

```csharp
public async Task<byte[]> ExportReportAsync(ReportType type, DateRange range)
{
    var reportData = type switch
    {
        ReportType.UserActivity => await GetUserActivityReportAsync(range),
        ReportType.DocumentUsage => await GetDocumentUsageReportAsync(range),
        ReportType.CostAnalysis => await GetCostAnalysisReportAsync(range),
        _ => throw new ArgumentException("Unknown report type")
    };
    
    return type switch
    {
        ReportFormat.Csv => ExportToCsv(reportData),
        ReportFormat.Excel => ExportToExcel(reportData),
        ReportFormat.Pdf => ExportToPdf(reportData),
        _ => throw new ArgumentException("Unknown format")
    };
}
```
