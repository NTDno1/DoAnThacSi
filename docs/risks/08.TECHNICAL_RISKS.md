```csharp
public class OllamaFallbackService : ILLMService
{
    private readonly OllamaService _ollamaService;
    
    public async Task<ChatResponse> ChatAsync(ChatRequest request)
    {
        try
        {
            return await _ollamaService.ChatAsync(request);
        }
        catch (OllamaConnectionException)
        {
            return GenerateBasicFallbackResponse(request);
        }
        catch (OutOfMemoryException)
        {
            await ClearOllamaCacheAsync();
            return await _ollamaService.ChatAsync(request);
        }
    }
    
    private ChatResponse GenerateBasicFallbackResponse(ChatRequest request)

        return new ChatResponse

            Confidence = ConfidenceLevel.VeryLow,
            IsFallback = true
        };
    }
}
```

**Code implementation**:

```csharp
public class RAGPipeline
{
    public async Task<RAGResponse> GenerateWithGrounding(
        string question, 
        List<RetrievedChunk> retrievedChunks)
    {
        // Bước 1: Kiểm tra độ phù hợp của retrieved chunks
        var minSimilarityThreshold = 0.5;
        var relevantChunks = retrievedChunks
            .Where(c => c.SimilarityScore >= minSimilarityThreshold)
            .ToList();

        // Bước 2: Nếu không có chunk nào đủ phù hợp, fallback
        if (!relevantChunks.Any())
        {
            return new RAGResponse
            {
                Answer = "Tôi không tìm thấy thông tin liên quan trong tài liệu.",
                Confidence = ConfidenceLevel.Low,
                Sources = new List<Source>()
            };
        }

        // Bước 3: Build prompt với citation requirement
        var prompt = BuildPromptWithCitation(question, relevantChunks);

        // Bước 4: Generate với hướng dẫn citation
        var response = await _llmService.GenerateAsync(prompt);

        // Bước 5: Verify citation exists in context
        var verifiedResponse = await VerifyCitationsAsync(response, relevantChunks);

        return verifiedResponse;
    }
}
```

#### 2.4.3 Confidence Scoring

```python
# Confidence scoring dựa trên similarity score
def calculate_confidence(similarity_scores: List[float], 
                         retrieved_count: int) -> dict:
    """
    Tính confidence score dựa trên:
    - Average similarity của top chunks
    - Số lượng chunks retrieved
    - Spread của similarity scores
    """
    
    if not similarity_scores:
        return {"level": "very_low", "score": 0.0, "reason": "no_results"}
    
    avg_similarity = sum(similarity_scores) / len(similarity_scores)
    max_similarity = max(similarity_scores)
    min_similarity = min(similarity_scores)
    
    # Confidence levels
    if max_similarity > 0.85 and avg_similarity > 0.7:
        level = "high"
        score = 0.9
    elif max_similarity > 0.7 and avg_similarity > 0.5:
        level = "medium"
        score = 0.7
    elif max_similarity > 0.5:
        level = "low"
        score = 0.4
    else:
        level = "very_low"
        score = 0.1
    
    return {
        "level": level,
        "score": score,
        "max_similarity": max_similarity,
        "avg_similarity": avg_similarity,
        "chunk_count": retrieved_count
    }
```

#### 2.4.5 Evaluation Framework với RAGAS

```python
# RAGAS metrics cho đánh giá chất lượng RAG
from ragas import evaluate
from ragas.metrics import (
    faithfulness,
    answer_relevancy,
    context_precision,
    context_recall
)

def evaluate_rag_system(test_dataset):
    """
    Đánh giá hệ thống RAG sử dụng RAGAS metrics:
    - Faithfulness: Mức độ câu trả lời trung thành với context
    - Answer Relevancy: Mức độ câu trả lời liên quan đến câu hỏi
    - Context Precision: Độ chính xác của context được retrieve
    - Context Recall: Độ phủ của context
    """
    
    metrics = [
        faithfulness,       # Target: > 0.8
        answer_relevancy,   # Target: > 0.75
        context_precision,  # Target: > 0.7
        context_recall      # Target: > 0.7
    ]
    
    results = evaluate(
        dataset=test_dataset,
        metrics=metrics
    )
    
    return results

# Benchmark thresholds

    "answer_relevancy": 0.75,  # 75% câu trả lời phải relevant
```

## 3. OCR ERROR RISK

### 3.1 Mô tả Rủi ro
OCR (Optical Character Recognition) có thể trích xuất text sai từ tài liệu scan, đặc biệt với:
- Font Tiếng Việt phức tạp (VNI, TCVN3)
- Tài liệu chất lượng thấp
- Scan nghiêng hoặc bị nhiễu
- Bảng biểu phức tạp

### 3.2 Mức độ nghiêm trọng: **HIGH**
- **Khả năng xảy ra**: MEDIUM (30-40%)
- **Tác động**: HIGH
- **Ưu tiên xử lý**: P1

### 3.3 Tác động cụ thể

| Tác động | Mô tả | Độ nghiêm trọng |
|----------|-------|-----------------|
| Search sai | Người dùng không tìm thấy nội dung đúng | HIGH |
| Chatbot sai | Trả lời dựa trên text sai | HIGH |
| UX kém | Trải nghiệm tìm kiếm không nhất quán | MEDIUM |
| Data corruption | Text sai được lưu vào database | HIGH |

#### 3.4.2 Pre-processing Pipeline

```python
from PIL import Image, ImageEnhance, ImageFilter
import cv2
import numpy as np

class ImagePreprocessor:
    """
    Pre-processing pipeline cho OCR:
    1. Convert to grayscale
    2. Deskew (chỉnh độ nghiêng)
    3. Denoise (khử nhiễu)
    4. Enhance contrast
    5. Binarization
    """
    
    def preprocess(self, image_path: str) -> Image:
        """Pre-process image trước khi OCR"""
        
        # Bước 1: Load và convert
        img = cv2.imread(image_path)
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        
        # Bước 2: Deskew - phát hiện và chỉnh độ nghiêng
        coords = np.column_stack(np.where(gray > 0))
        angle = cv2.minAreaRect(coords)[-1]
        if angle < -45:
            angle = -(90 + angle)
        else:
            angle = -angle
        
        (h, w) = gray.shape[:2]
        center = (w // 2, h // 2)
        M = cv2.getRotationMatrix2D(center, angle, 1.0)
        rotated = cv2.warpAffine(gray, M, (w, h), 
                                  flags=cv2.INTER_CUBIC,
                                  borderMode=cv2.BORDER_REPLICATE)
        
        # Bước 3: Denoise
        denoised = cv2.fastNlMeansDenoising(rotated, None, 10, 7, 21)
        
        # Bước 4: Enhance contrast
        enhancer = ImageEnhance.Contrast(Image.fromarray(denoised))
        enhanced = enhancer.enhance(1.5)
        
        # Bước 5: Binarization
        _, binary = cv2.threshold(denoised, 0, 255, 
                                  cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        
        return Image.fromarray(binary)
```

```python
from sentence_transformers import CrossEncoder
from rank_bm25 import BM25Okapi
import numpy as np

class HybridSearchEngine:
    """
    Hybrid Search kết hợp:
    - Vector search (semantic similarity)
    - BM25 (keyword matching)
    - RRF (Reciprocal Rank Fusion) cho combining
    """
    
    def __init__(self, vector_store, chunks):
        self.vector_store = vector_store
        self.chunks = chunks
        
        # Setup BM25
        tokenized_chunks = [chunk.text.split() for chunk in chunks]
        self.bm25 = BM25Okapi(tokenized_chunks)
        
        # Setup Cross-encoder cho reranking
        self.cross_encoder = CrossEncoder('cross-encoder/ms-marco-MiniLM-L-6-v2')
    
    def search(self, query: str, top_k: int = 10) -> List[SearchResult]:
        """Hybrid search với RRF fusion"""
        
        # Bước 1: Vector search
        vector_results = self.vector_store.similarity_search(
            query, 
            k=50  # Lấy nhiều hơn để combine
        )
        
        # Bước 2: BM25 keyword search
        bm25_scores = self.bm25.get_scores(query.split())
        bm25_results = sorted(
            range(len(bm25_scores)), 
            key=lambda i: bm25_scores[i], 
            reverse=True
        )[:50]
        
        # Bước 3: RRF Fusion
        rrf_scores = self._reciprocal_rank_fusion(
            vector_results, 
            bm25_results,
            k=60  # RRF parameter
        )
        
        # Bước 4: Lấy top candidates
        top_candidates = sorted(
            rrf_scores.items(), 
            key=lambda x: x[1], 
            reverse=True
        )[:top_k * 2]  # Lấy nhiều hơn để rerank
        
        # Bước 5: Cross-encoder reranking
        reranked = self._rerank_with_cross_encoder(
            query, 
            top_candidates
        )
        
        return reranked[:top_k]
    
    def _reciprocal_rank_fusion(
        self, 
        results_a: List, 
        results_b: List, 
        k: int = 60
    ) -> Dict[int, float]:
        """
        Reciprocal Rank Fusion (RRF)
        Combines multiple result sets using rank-based scoring
        """
        rrf_scores = {}
        
        # Add vector search scores (weight 0.7)
        for rank, result in enumerate(results_a):
            doc_id = result.chunk_id
            score = 1 / (k + rank + 1)
            rrf_scores[doc_id] = rrf_scores.get(doc_id, 0) + (0.7 * score)
        
        # Add BM25 scores (weight 0.3)
        for rank, doc_id in enumerate(results_b):
            score = 1 / (k + rank + 1)
            rrf_scores[doc_id] = rrf_scores.get(doc_id, 0) + (0.3 * score)
        
        return rrf_scores
    
    def _rerank_with_cross_encoder(
        self, 
        query: str, 
        candidates: List[Tuple[int, float]]
    ) -> List[SearchResult]:
        """Rerank sử dụng Cross-Encoder"""
        
        # Prepare pairs (query, document)
        pairs = []
        for doc_id, _ in candidates:
            chunk = self._get_chunk(doc_id)
            pairs.append((query, chunk.text))
        
        # Get cross-encoder scores
        cross_scores = self.cross_encoder.predict(pairs)
        
        # Combine với RRF scores
        final_scores = []
        for i, (doc_id, rrf_score) in enumerate(candidates):
            combined_score = 0.6 * cross_scores[i] + 0.4 * rrf_score
            final_scores.append((doc_id, combined_score))
        
        return sorted(final_scores, key=lambda x: x[1], reverse=True)
```

#### 4.3.3 MMR (Maximum Marginal Relevance)

```python
class DiversityAwareSearch:
    """
    MMR để đảm bảo diversity trong search results
    Tránh trả về nhiều kết quả quá giống nhau
    """
    
    def __init__(self, embedding_model):
        self.embedding_model = embedding_model
    
    def search_with_mmr(
        self, 
        query: str, 
        chunks: List[Chunk], 
        k: int = 10, 
        lambda_param: float = 0.5
    ) -> List[Chunk]:
        """
        MMR search
        lambda_param: Độ quan trọng của relevance vs diversity
        - 0: Chỉ diversity
        - 1: Chỉ relevance
        - 0.5: Cân bằng
        """
        
        query_embedding = self.embedding_model.encode(query)
        
        # Tính relevance scores
        relevance_scores = []
        for chunk in chunks:
            similarity = self._cosine_similarity(
                query_embedding, 
                chunk.embedding
            )
            relevance_scores.append((chunk, similarity))
        
        # Sort by relevance
        relevance_scores.sort(key=lambda x: x[1], reverse=True)
        
        # MMR selection
        selected = []
        remaining = [c for c, _ in relevance_scores]
        
        while len(selected) < k and remaining:
            best_score = -float('inf')
            best_chunk = None
            
            for chunk in remaining:
                relevance = self._cosine_similarity(
                    query_embedding, 
                    chunk.embedding
                )
                
                # Tính max similarity với selected chunks (diversity)
                if selected:
                    max_similarity = max(
                        self._cosine_similarity(chunk.embedding, s.embedding)
                        for s in selected
                    )
                else:
                    max_similarity = 0
                
                # MMR score
                mmr_score = (
                    lambda_param * relevance - 
                    (1 - lambda_param) * max_similarity
                )
                
                if mmr_score > best_score:
                    best_score = mmr_score
                    best_chunk = chunk
            
            if best_chunk:
                selected.append(best_chunk)
                remaining.remove(best_chunk)
        
        return selected
    
    def _cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))
```

#### 4.3.4 Evaluation Metrics

```python
from typing import List
import numpy as np

class SearchEvaluator:
    """
    Đánh giá search quality với các metrics:
    - NDCG@K (Normalized Discounted Cumulative Gain)
    - Precision@K
    - Recall@K
    - MRR (Mean Reciprocal Rank)
    """
    
    def calculate_ndcg(
        self, 
        predicted_scores: List[float], 
        relevance_labels: List[int], 
        k: int = 10
    ) -> float:
        """
        NDCG@K - đánh giá ranking quality
        Target: > 0.7
        """
        
        # Sắp xếp predicted scores
        sorted_indices = np.argsort(predicted_scores)[::-1][:k]
        
        # Tính DCG
        dcg = 0
        for i, idx in enumerate(sorted_indices):
            dcg += relevance_labels[idx] / np.log2(i + 2)
        
        # Tính IDCG (ideal DCG)
        ideal_sorted = sorted(relevance_labels, reverse=True)[:k]
        idcg = sum(
            rel / np.log2(i + 2) 
            for i, rel in enumerate(ideal_sorted)
        )
        
        # NDCG
        return dcg / idcg if idcg > 0 else 0
    
    def calculate_precision_at_k(
        self, 
        predicted: List[bool], 
        k: int = 10
    ) -> float:
        """Precision@K - tỷ lệ relevant trong top K"""
        return sum(predicted[:k]) / k
    
    def calculate_recall_at_k(
        self, 
        predicted: List[bool], 
        relevant_count: int, 
        k: int = 10
    ) -> float:
        """Recall@K - coverage của relevant items"""
        return sum(predicted[:k]) / relevant_count
    
    def calculate_mrr(self, ranked_results: List[bool]) -> float:
        """
        MRR - Mean Reciprocal Rank
        Trung bình của 1/rank của first relevant item
        """
        for i, is_relevant in enumerate(ranked_results):
            if is_relevant:
                return 1 / (i + 1)
        return 0
```

```csharp
public class CacheService
{
    private readonly IConnectionMultiplexer _redis;
    private readonly IDatabase _db;
    private readonly IMemoryCache _memoryCache;
    
    // Cache keys
    private const string EmbeddingPrefix = "embedding:";
    private const string SearchPrefix = "search:";
    private const string DocumentPrefix = "doc:";
    
    // TTLs
    private static readonly TimeSpan EmbeddingTTL = TimeSpan.FromMinutes(5);
    private static readonly TimeSpan SearchTTL = TimeSpan.FromMinutes(30);
    private static readonly TimeSpan DocumentTTL = TimeSpan.FromHours(1);
    
    public CacheService(IConnectionMultiplexer redis, IMemoryCache memoryCache)
    {
        _redis = redis;
        _db = redis.GetDatabase();
        _memoryCache = memoryCache;
    }
    
    // L1: Memory cache cho embeddings
    public async Task<float[]?> GetEmbeddingAsync(string text)
    {
        var key = $"{EmbeddingPrefix}{text.GetHashCode()}";
        
        // Check L1 (memory)
        if (_memoryCache.TryGetValue(key, out float[]? cached))
        {
            return cached;
        }
        
        // Check L2 (Redis)
        var redisValue = await _db.StringGetAsync(key);
        if (redisValue.HasValue)
        {
            var embedding = JsonSerializer.Deserialize<float[]>(redisValue!);
            // Populate L1
            _memoryCache.Set(key, embedding, EmbeddingTTL);
            return embedding;
        }
        
        return null;
    }
    
    public async Task SetEmbeddingAsync(string text, float[] embedding)
    {
        var key = $"{EmbeddingPrefix}{text.GetHashCode()}";
        
        // Set L1
        _memoryCache.Set(key, embedding, EmbeddingTTL);
        
        // Set L2
        var value = JsonSerializer.Serialize(embedding);
        await _db.StringSetAsync(key, value, EmbeddingTTL);
    }
    
    // Search results cache
    public async Task<List<SearchResult>?> GetSearchResultsAsync(
        string query, 
        Dictionary<string, string> filters)
    {
        var key = GenerateSearchCacheKey(query, filters);
        
        // Check Redis
        var cached = await _db.StringGetAsync(key);
        if (cached.HasValue)
        {
            return JsonSerializer.Deserialize<List<SearchResult>>(cached!);
        }
        
        return null;
    }
    
    public async Task SetSearchResultsAsync(
        string query, 
        Dictionary<string, string> filters,
        List<SearchResult> results)
    {
        var key = GenerateSearchCacheKey(query, filters);
        var value = JsonSerializer.Serialize(results);
        
        await _db.StringSetAsync(key, value, SearchTTL);
    }
```

```csharp
public class DocumentProcessingBackgroundService : BackgroundService

        }
    }
```

```csharp
// Authorization service interface
public interface IAuthorizationService
{
    Task<bool> CanAccessDocumentAsync(Guid userId, Guid documentId);
    Task<bool> CanAccessChunkAsync(Guid userId, Guid chunkId);
    Task<bool> CanSearchAsync(Guid userId);
    Task<bool> CanChatAsync(Guid userId);
}

public class DocumentAuthorizationService : IAuthorizationService
{
    private readonly ApplicationDbContext _context;
    
    public async Task<bool> CanAccessDocumentAsync(Guid userId, Guid documentId)
    {
        var document = await _context.Documents
            .AsNoTracking()
            .FirstOrDefaultAsync(d => d.Id == documentId);
        
        if (document == null)
            return false;
        
        // Check ownership
        if (document.OwnerId == userId)
            return true;
        
        // Check if public
        if (document.IsPublic)
            return true;
        
        // Check admin
        var user = await _context.Users
            .AsNoTracking()
            .FirstOrDefaultAsync(u => u.Id == userId);
        
        if (user?.Role == UserRole.Admin)
            return true;
        
        // Check department access
        var departmentAccess = await _context.UserDepartments
            .AnyAsync(ud => ud.UserId == userId 
                && ud.DepartmentId == document.DepartmentId);
        
        return departmentAccess;
    }
    
    public async Task<bool> CanAccessChunkAsync(Guid userId, Guid chunkId)
    {
        var chunk = await _context.DocumentChunks
            .AsNoTracking()
            .Include(c => c.Document)
            .FirstOrDefaultAsync(c => c.Id == chunkId);
        
        if (chunk == null)
            return false;
        
        return await CanAccessDocumentAsync(userId, chunk.DocumentId);
    }
}
```

```csharp
public class SecureSearchService
{
    private readonly ApplicationDbContext _context;
    private readonly IAuthorizationService _authService;
    
    public async Task<SearchResponse> SearchAsync(
        SearchQuery query, 
        Guid userId)
    {
        // Get user's accessible department IDs
        var accessibleDepartments = await GetAccessibleDepartmentsAsync(userId);
        
        // Build query với permission filter
        var searchQuery = _context.DocumentChunks
            .AsNoTracking()
            .Include(c => c.Document)
            .Where(c => c.Document.IsActive)
            .Where(c => 
                // Public documents
                c.Document.IsPublic
                // OR user's own documents
                || c.Document.OwnerId == userId
                // OR department documents
                || accessibleDepartments.Contains(c.Document.DepartmentId)
                // OR admin
                || _context.Users.Any(u => 
                    u.Id == userId && u.Role == UserRole.Admin)
            );
        
        // Add semantic search filter
        searchQuery = searchQuery
            .OrderBy(c => c.ContentEmbedding.MaxCosineDistance(query.Embedding))
            .Take(query.Limit);
        
        var results = await searchQuery.ToListAsync();
        
        // Final permission check (defense in depth)
        var authorizedResults = results
            .Where(r => _authService.CanAccessDocumentAsync(userId, r.DocumentId).Result)
            .Select(r => MapToSearchResult(r, query))
            .ToList();
        
        return new SearchResponse
        {
            Results = authorizedResults,
            TotalCount = authorizedResults.Count,
            UserId = userId,
            Timestamp = DateTime.UtcNow
        };
    }
}
```
